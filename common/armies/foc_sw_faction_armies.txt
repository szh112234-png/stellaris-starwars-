# V119 - Knights of Ren

# the formation once Ben Solo's fall has occurred globally.
# =============================================================================
foc_sw_knights_of_ren = {
    damage = 3.40
    health = 2.40
    morale = 3.20
    morale_damage = 3.60
    collateral_damage = 0.50
    war_exhaustion = 2.00
    time = 240
    has_species = no
    pop_limited = no
    icon = GFX_foc_sw_army_knights_of_ren

    resources = {
        category = armies
        cost = { minerals = 650 }
        upkeep = { energy = 6 }
    }

    potential = {
        owner = {
            OR = {
                has_country_flag = foc_khan_first_order_country
                has_country_flag = fop_player_first_order
                has_country_flag = foc_republic_origin_became_first_order
            }

            # Player timing follows the dedicated Knights-of-Ren story event.
            # AI First Order gets the same unit once Ben Solo has fallen.
            OR = {
                has_modifier = foc_saga_knights_of_ren
                AND = {
                    is_ai = yes
                    has_global_flag = foc_saga_ben_solo_fallen
                }
            }

            # Once Ben has returned to the light, no new Knights are raised.
            NOT = { has_global_flag = foc_saga_ben_solo_returned }

            NOT = { has_country_flag = foc_khan_final_order_country }
            NOT = { has_country_flag = fop_player_final_order }
            NOT = { has_country_flag = foc_republic_origin_empire_country }
            NOT = { has_country_flag = foc_sw_imperial_remnant_country }
            NOT = { has_country_flag = foc_sw_new_republic_country }
            NOT = { has_country_flag = foc_khan_rebel_alliance_country }
            NOT = { has_country_flag = foc_khan_first_order_resistance_country }
        }

        planet = {
            has_active_building = building_foc_sw_first_order_training_center
        }
    }

    allow = {
        has_active_building = building_foc_sw_first_order_training_center

        # The Knights are a single order, not a mass-produced army branch.
        custom_tooltip = {
            text = foc_sw_knights_of_ren_limit_tt
            owner = {
                check_variable = {
                    which = "foc_sw_knights_of_ren_count"
                    value < 1
                }
            }
        }
    }

    on_queued = {
        owner = {
            change_variable = {
                which = "foc_sw_knights_of_ren_count"
                value = 1
            }
        }
    }

    on_unqueued = {
        owner = {
            change_variable = {
                which = "foc_sw_knights_of_ren_count"
                value = -1
            }
        }
    }

    army_modifier = {
        army_morale_damage_mult = 0.25
        army_damage_mult = 0.10
        army_disengage_chance_mult = 0.20
    }

    ai_weight = {
        base = 200
        modifier = {
            set = 0
            check_variable = {
                which = "foc_sw_knights_of_ren_count"
                value >= 1
            }
        }
    }
}
