# Jedi field formations for the Republic origin. The stat profile follows the
# Grand Army of the Republic pack's own Jedi armies so the two feel identical,
# but the keys, the building requirement and the ownership condition all belong
# to this origin, so the pack's definitions are left untouched.
#
# They exist only while the Temple stands. Order 66 demolishes it, which ends
# recruitment on the same night the Order itself ends.

foc_republic_origin_jedi_padawans = {
    damage = 1.3
    health = 1.2
    morale = 1.5
    morale_damage = 1.2
    collateral_damage = 0.5
    war_exhaustion = 1.2
    time = 100
    icon = GFX_foc_army_type_jedi

    resources = {
        category = armies
        cost = { minerals = 250 }
        upkeep = { energy = 1.5 }
    }

    allow = {
        has_active_building = building_foc_republic_origin_jedi_temple
    }

    potential = {
        from = {
            NOR = {
                has_trait = "trait_mechanical"
                has_trait = "trait_machine_unit"
                is_sapient = no
            }
        }
        planet = {
            has_active_building = building_foc_republic_origin_jedi_temple
        }
        owner = {
            has_country_flag = foc_republic_origin_jedi_council_created
            NOT = { has_country_flag = foc_republic_origin_jedi_purged }
            NOT = { has_authority = auth_machine_intelligence }
        }
    }

    army_modifier = {
        army_morale = 0.05
    }
}
