# FOC 3.07 - direct-output infrastructure for the New Republic and Imperial Remnant.
# Production is conditional on the scripted Fallen Empire flag, so conquered
# buildings do not give ordinary empires the full ancient-economy output.
#
# FOC V28: foc_sw_fe_infrastructure_heir marks the story successors that keep
# the ancient economy running - the First Order and its Final Order form, which
# inherit these worlds by script rather than by ordinary conquest. Upkeep is now
# gated by the same condition: a country that receives no output from a captured
# ancient building no longer pays its running cost either. Without this, losing
# the scripted Fallen Empire flag stopped all production while leaving every
# upkeep line active, which drained the successor's stockpiles to zero.

building_foc_sw_fe_civil_nexus = {
    base_buildtime = 480
    can_build = no
    can_demolish = yes
    planet_limit = 1
    icon = building_fe_administration_2
    category = government
    building_sets = { government urban fallen_empire }

    potential = { always = yes }
    destroy_trigger = { always = no }

    triggered_planet_modifier = {
        potential = {
            exists = owner
            owner = {
                OR = {
                    has_country_flag = foc_sw_scripted_fallen_empire
                    has_country_flag = foc_sw_fe_infrastructure_heir
                }
            }
        }
        modifier = {
            planet_housing_add = 500
            planet_amenities_add = 500
            planet_stability_add = 20
            planet_crime_add = -100
            planet_jobs_produces_mult = 0.50
        }
    }

    resources = {
        category = planet_buildings
        produces = {
            trigger = {
                exists = owner
                owner = {
                    OR = {
                        has_country_flag = foc_sw_scripted_fallen_empire
                        has_country_flag = foc_sw_fe_infrastructure_heir
                    }
                }
            }
            unity = 250
        }
    }
}
