# Colony/planet scope, expanded inline into FOC callers.
# Preserve a habitat's own capital/district set instead of applying terrestrial setup.
if = {
    limit = { uses_habitat_capitals = yes }
    if = {
        limit = {
            NOT = { has_building = building_hab_capital }
            NOT = { has_building = building_hab_major_capital }
            free_building_slots > 0
            exists = owner
            owner = { is_regular_empire = yes is_fallen_empire = no }
        }
        add_building = building_hab_capital
    }
}
else_if = {
    # These worlds receive the existing FOC ancient infrastructure immediately afterwards.
    limit = { exists = owner owner = { is_fallen_empire = yes has_country_flag = foc_sw_scripted_fallen_empire } }
}
else = {
    generate_start_buildings_and_districts = yes
}
