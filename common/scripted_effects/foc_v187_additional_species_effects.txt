# V187 - additive species only. Existing FOC native species are intentionally untouched.

foc_v187_seed_twilek = {
    if = {
        limit = {
            is_colony = yes
            NOT = { has_planet_flag = foc_v187_twilek_seeded }
        }

        create_species = {
            name = foc_v187_twilek_name
            plural = foc_v187_twilek_plural
            adjective = foc_v187_twilek_adjective
            name_list = "rebels"
            class = "HUM"
            portrait = "foc_v187_twilek_species_01"
            homeworld = this
            traits = {
                trait = "trait_adaptive"
                trait = "trait_charismatic"
                trait = "trait_sedentary"
            }
        }

        last_created_species = {
            set_species_flag = foc_v187_twilek_species
            save_global_event_target_as = foc_v187_twilek_species
            set_citizenship_type = { country = prev.owner type = citizenship_full cooldown = no }
            set_military_service_type = { country = prev.owner type = military_service_full cooldown = no }
            set_population_controls = { country = prev.owner type = population_control_no cooldown = no }
            set_migration_controls = { country = prev.owner type = migration_control_no cooldown = no }
            set_colonization_controls = { country = prev.owner type = colonization_control_no cooldown = no }
        }

        while = {
            count = 3
            create_pop_group = {
                size = 100
                species = last_created_species
            }
        }

        set_planet_flag = foc_v187_twilek_seeded
    }
}
