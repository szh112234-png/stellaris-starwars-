# V135 - AI-only strategic shipyard support for Rebel states.
# Country scope. These are annual support fleets and intentionally do not apply to human players.

# Mon Cala: extra annual fleet of exactly 20 dedicated Rebel main-line MC80s.
foc_v135_spawn_mon_cala_20_mc80_line = {
    if = {
        limit = { NOT = { has_global_flag = foc_v211_ai_support_rebel_disabled } }
        event_target:foc_swmap_mon_cala_system = {
            if = {
                limit = {
                    exists = space_owner
                    space_owner = { is_same_value = root }
                    NOT = {
                        any_system_planet = {
                            is_colony = yes
                            NOT = { owner = { is_same_value = root } }
                        }
                    }
                }
                create_fleet = {
                    name = "FOC_V135_MON_CALA_MC80_SUPPORT"
                    settings = {
                        can_upgrade = no
                        can_disband = yes
                        can_change_composition = no
                        spawn_debris = yes
                        ai_ignore_strength = yes
                    }
                    effect = {
                        set_owner = root
                        foc_v211_create_scaled_rebel_20_mc80_line = yes
                        set_location = { target = prev distance = 45 angle = random }
                        set_aggro_range_measure_from = self
                        set_aggro_range = 9999
                        set_fleet_stance = aggressive
                        set_formation_scale = 2
                        set_fleet_flag = foc_republic_origin_rebellion_fleet
                        set_fleet_flag = foc_v135_mon_cala_support_fleet
                    }
                }
            }
        }
    }
}

# Kuat: annual fleet of exactly 15 Nebula-class warships.
foc_v135_spawn_kuat_15_nebula = {
    if = {
        limit = { NOT = { has_global_flag = foc_v211_ai_support_rebel_disabled } }
        random_system = {
            limit = {
                has_star_flag = foc_sw_kuat_home
                space_owner? = { is_same_value = root }
                NOT = {
                    any_system_planet = {
                        is_colony = yes
                        NOT = { owner = { is_same_value = root } }
                    }
                }
            }
            create_fleet = {
                name = "FOC_V135_KUAT_NEBULA_SUPPORT"
                settings = {
                    can_upgrade = no
                    can_disband = yes
                    can_change_composition = no
                    spawn_debris = yes
                    ai_ignore_strength = yes
                }
                effect = {
                    set_owner = root
                    foc_v211_create_scaled_rebel_15_nebulas = yes
                    set_location = { target = prev distance = 50 angle = random }
                    set_aggro_range_measure_from = self
                    set_aggro_range = 9999
                    set_fleet_stance = aggressive
                    set_formation_scale = 2
                    set_fleet_flag = foc_republic_origin_rebellion_fleet
                    set_fleet_flag = foc_v135_kuat_nebula_support_fleet
                }
            }
        }
    }
}
