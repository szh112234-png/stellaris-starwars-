# V17b: every max() comparison now subtracts into foc_v235_cmp and tests it
# against a literal. Reading a variable on the right-hand side of a trigger
# was the thing logging "Variable foc_v235_need_common is not set"; the same
# read works fine on the effect side, which is where it now happens.
# V235 player treasury succession. Native APIs follow the supplied vanilla
# nomads_effects.txt and 000_how_to_use_variables_in_script.txt.
# Same-country succession restores ONLY a shortfall, never credits a second copy.
# Resource quantities are captured before old era storage modifiers are removed.

foc_v235_ensure_resistance_storage = {
    if = {
        limit = { NOT = { has_modifier = foc_v235_resistance_storage } }
        add_modifier = { modifier = foc_v235_resistance_storage days = -1 }
    }
}

# Country scope. Callers fill the three requested floors; this only raises them.
foc_v235_raise_treasury_floors = {
    hidden_effect = {
        # Keep every stored resource protected even if a modifier replacement causes
        # an immediate engine capacity refresh. Never remove the old floor unbridged.
        set_variable = { which = foc_v235_bridge_capacity value = foc_v235_need_common }
        set_variable = { which = foc_v235_cmp value = foc_v235_need_artifact }
        subtract_variable = { which = foc_v235_cmp value = foc_v235_bridge_capacity }
        if = {
            limit = { check_variable = { which = foc_v235_cmp value > 0 } }
            set_variable = { which = foc_v235_bridge_capacity value = foc_v235_need_artifact }
        }
        set_variable = { which = foc_v235_cmp value = foc_v235_need_thread }
        subtract_variable = { which = foc_v235_cmp value = foc_v235_bridge_capacity }
        if = {
            limit = { check_variable = { which = foc_v235_cmp value > 0 } }
            set_variable = { which = foc_v235_bridge_capacity value = foc_v235_need_thread }
        }
        if = {
            limit = { check_variable = { which = foc_v235_bridge_capacity value > 0 } }
            add_modifier = { modifier = foc_v235_treasury_capacity_bridge days = -1 multiplier = foc_v235_bridge_capacity }
        }
        # V17: no zero initialiser; seed the floor from the already-set need value.
        if = {
            limit = { NOT = { is_variable_set = foc_v235_common_floor } }
            set_variable = { which = foc_v235_common_floor value = foc_v235_need_common }
            remove_modifier = foc_v235_inherited_treasury_storage
        }
        else_if = {
            limit = { check_variable = { which = foc_v235_need_common value > foc_v235_common_floor } }
            set_variable = { which = foc_v235_common_floor value = foc_v235_need_common }
            remove_modifier = foc_v235_inherited_treasury_storage
        }
        if = {
            limit = {
                check_variable = { which = foc_v235_common_floor value > 0 }
                NOT = { has_modifier = foc_v235_inherited_treasury_storage }
            }
            add_modifier = { modifier = foc_v235_inherited_treasury_storage days = -1 multiplier = foc_v235_common_floor }
        }
        # V17: no zero initialiser; seed the floor from the already-set need value.
        if = {
            limit = { NOT = { is_variable_set = foc_v235_artifact_floor } }
            set_variable = { which = foc_v235_artifact_floor value = foc_v235_need_artifact }
            remove_modifier = foc_v235_inherited_artifact_storage
        }
        else_if = {
            limit = { check_variable = { which = foc_v235_need_artifact value > foc_v235_artifact_floor } }
            set_variable = { which = foc_v235_artifact_floor value = foc_v235_need_artifact }
            remove_modifier = foc_v235_inherited_artifact_storage
        }
        if = {
            limit = {
                check_variable = { which = foc_v235_artifact_floor value > 0 }
                NOT = { has_modifier = foc_v235_inherited_artifact_storage }
            }
            add_modifier = { modifier = foc_v235_inherited_artifact_storage days = -1 multiplier = foc_v235_artifact_floor }
        }
        # V17: no zero initialiser; seed the floor from the already-set need value.
        if = {
            limit = { NOT = { is_variable_set = foc_v235_thread_floor } }
            set_variable = { which = foc_v235_thread_floor value = foc_v235_need_thread }
            remove_modifier = foc_v235_inherited_thread_storage
        }
        else_if = {
            limit = { check_variable = { which = foc_v235_need_thread value > foc_v235_thread_floor } }
            set_variable = { which = foc_v235_thread_floor value = foc_v235_need_thread }
            remove_modifier = foc_v235_inherited_thread_storage
        }
        if = {
            limit = {
                check_variable = { which = foc_v235_thread_floor value > 0 }
                NOT = { has_modifier = foc_v235_inherited_thread_storage }
            }
            add_modifier = { modifier = foc_v235_inherited_thread_storage days = -1 multiplier = foc_v235_thread_floor }
        }
        remove_modifier = foc_v235_treasury_capacity_bridge
        clear_variable = foc_v235_bridge_capacity
    }
}

# Source country scope. TARGET is a real player successor. Called BEFORE donor
# destruction. In-place calls and AI recipients are rejected. Never a monthly grant.

        save_event_target_as = foc_v235_treasury_donor
        if = {
            limit = {
                exists = $TARGET$
                NOT = { has_country_flag = foc_v235_treasury_transferred }
                $TARGET$ = {
                    is_ai = no
                    NOT = { is_same_value = event_target:foc_v235_treasury_donor }
                }
            }
            $TARGET$ = { save_event_target_as = foc_v235_treasury_recipient }
            export_resource_stockpile_to_variable = { resource = energy variable = foc_v235_donor_energy }
            export_resource_stockpile_to_variable = { resource = minerals variable = foc_v235_donor_minerals }
            export_resource_stockpile_to_variable = { resource = food variable = foc_v235_donor_food }
            export_resource_stockpile_to_variable = { resource = trade variable = foc_v235_donor_trade }
            export_resource_stockpile_to_variable = { resource = consumer_goods variable = foc_v235_donor_consumer_goods }
