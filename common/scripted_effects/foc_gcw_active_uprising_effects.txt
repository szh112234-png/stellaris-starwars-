# Espionage-operation scope. Clears the persistent target and owner lock.
foc_gcw_active_uprising_cleanup = {
    if = {
        limit = {
            OR = {
                NOT = { exists = event_target:foc_v232q_active_coup_operation }
                is_same_value = event_target:foc_v232q_active_coup_operation
            }
        }
        if = {
            limit = { exists = event_target:foc_gcw_active_uprising_target }
            event_target:foc_gcw_active_uprising_target = { remove_carrier_flag = foc_gcw_active_uprising_targeted }
            clear_global_event_target = foc_gcw_active_uprising_target
        }
        if = {
            limit = { exists = event_target:foc_gcw_active_uprising_target_system }
            clear_global_event_target = foc_gcw_active_uprising_target_system
        }
        foc_gcw_active_uprising_clear_candidates = yes
        if = {
            limit = { exists = event_target:foc_v232q_active_coup_operation }
            clear_global_event_target = foc_v232q_active_coup_operation
        }
        owner = {
            remove_country_flag = foc_gcw_active_uprising_in_progress
            remove_country_flag = foc_gcw_active_uprising_starting
        }
    }
}

# Country scope. Only orphaned state is released; active operations stay intact.
foc_v232q_cleanup_orphaned_coup = {
    if = {
        limit = { NOT = { any_espionage_operation = { has_espionage_type = operation_foc_gcw_instigate_system_uprising } } }
        remove_country_flag = foc_gcw_active_uprising_in_progress
        remove_country_flag = foc_gcw_active_uprising_starting
        if = {
            limit = {
                NOT = { any_country = { any_espionage_operation = { has_espionage_type = operation_foc_gcw_instigate_system_uprising } } }
            }
            if = {
                limit = { exists = event_target:foc_gcw_active_uprising_target }
                event_target:foc_gcw_active_uprising_target = { remove_carrier_flag = foc_gcw_active_uprising_targeted }
                clear_global_event_target = foc_gcw_active_uprising_target
            }
            if = {
                limit = { exists = event_target:foc_gcw_active_uprising_target_system }
                clear_global_event_target = foc_gcw_active_uprising_target_system
            }
            if = {
                limit = { exists = event_target:foc_v232q_active_coup_operation }
                clear_global_event_target = foc_v232q_active_coup_operation
            }
            foc_gcw_active_uprising_clear_candidates = yes
        }
    }
}
