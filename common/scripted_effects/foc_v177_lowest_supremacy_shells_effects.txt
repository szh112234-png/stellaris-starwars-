# Ship scope. Preserve ship object, carried colony, modules and story flags.
foc_v177_sync_supremacy_shell_ship = {
    if = {
        limit = {
            owner = { is_ai = no }
            OR = {
                is_ship_size = sai_supremacy_arkship_tier_1
                is_ship_size = sai_supremacy_arkship_tier_2
                AND = {
                    is_ship_size = sai_supremacy_arkship_tier_3
                    OR = {
                        has_ship_flag = foc_v177_supremacy_empty_shell
                        NOR = { has_component = SAI_SUPREMACY_SG_REACTOR_FIRE_CONTROL has_component = SAI_SUPREMACY_SG_REACTOR_FIRE_CONTROL_FOC_FIXED_LOWEST
                            has_component = FOC_V1718_EMP_REACTOR_4
                            has_component = FOC_V1718_EMP_REACTOR_4_LOWEST
                            has_component = FOC_V1718_EMP_REACTOR_5
                            has_component = FOC_V1718_EMP_REACTOR_5_LOWEST }
                    }
                }
                is_ship_size = sai_supremacy_arkship_tier_1_shell_foc_lowest
                is_ship_size = sai_supremacy_arkship_tier_2_shell_foc_lowest
                is_ship_size = sai_supremacy_arkship_tier_3_shell_foc_lowest
            }
        }
        set_ship_flag = foc_v177_supremacy_empty_shell
        if = {
            limit = { has_global_flag = foc_khan_strength_lowest }
            if = {
                limit = { has_modifier = sai_supremacy_arkship_stats }
                remove_modifier = sai_supremacy_arkship_stats
                set_ship_flag = foc_v177_shell_restore_stats
            }
            if = {
                limit = { has_modifier = foc_lowest_norm_sai_supremacy_t1 }
                remove_modifier = foc_lowest_norm_sai_supremacy_t1
            }
            if = {
                limit = { has_modifier = foc_lowest_norm_sai_supremacy_t2 }
                remove_modifier = foc_lowest_norm_sai_supremacy_t2
            }
            if = {
                limit = { has_modifier = foc_lowest_norm_sai_supremacy }
                remove_modifier = foc_lowest_norm_sai_supremacy
            }
            if = {
                limit = { is_ship_size = sai_supremacy_arkship_tier_1 }
                starbase = { set_starbase_size = sai_supremacy_arkship_tier_1_shell_foc_lowest }
                set_ship_design = { design = NAME_SAI_SUPREMACY_ARKSHIP_TIER_1_SHELL_FOC_LOWEST }
                set_graphical_culture = military_arkship
            }
            else_if = {
                limit = { is_ship_size = sai_supremacy_arkship_tier_2 }
                starbase = { set_starbase_size = sai_supremacy_arkship_tier_2_shell_foc_lowest }
                set_ship_design = { design = NAME_SAI_SUPREMACY_ARKSHIP_TIER_2_SHELL_FOC_LOWEST }
                set_graphical_culture = military_arkship
            }
            else_if = {
                limit = { is_ship_size = sai_supremacy_arkship_tier_3 }
                starbase = { set_starbase_size = sai_supremacy_arkship_tier_3_shell_foc_lowest }
                set_ship_design = { design = NAME_SAI_SUPREMACY_ARKSHIP_TIER_3_SHELL_FOC_LOWEST }
                set_graphical_culture = military_arkship
            }
        }
        else = {
            if = {
                limit = { is_ship_size = sai_supremacy_arkship_tier_1_shell_foc_lowest }
                starbase = { set_starbase_size = sai_supremacy_arkship_tier_1 }
                if = {
                    limit = { has_global_flag = foc_khan_strength_lowest }
                    set_ship_design = { design = "NAME_SAI_SUPREMACY_ARKSHIP_TIER_1_V1718E_LOWEST" }
                }
                else = {
                    set_ship_design = { design = NAME_SAI_SUPREMACY_ARKSHIP_TIER_1 }
                }
                set_graphical_culture = military_arkship
            }
            else_if = {
                limit = { is_ship_size = sai_supremacy_arkship_tier_2_shell_foc_lowest }
                starbase = { set_starbase_size = sai_supremacy_arkship_tier_2 }
                if = {
                    limit = { has_global_flag = foc_khan_strength_lowest }
                    set_ship_design = { design = "NAME_SAI_SUPREMACY_ARKSHIP_TIER_2_V1718E_LOWEST" }
                }
                else = {
                    set_ship_design = { design = NAME_SAI_SUPREMACY_ARKSHIP_TIER_2 }
                }
                set_graphical_culture = military_arkship
            }
            else_if = {
                limit = { is_ship_size = sai_supremacy_arkship_tier_3_shell_foc_lowest }
                starbase = { set_starbase_size = sai_supremacy_arkship_tier_3 }
                if = {
                    limit = { has_global_flag = foc_khan_strength_lowest }
                    set_ship_design = { design = "NAME_SAI_SUPREMACY_ARKSHIP_TIER_3_V1718E_LOWEST" }
                }
                else = {
                    set_ship_design = { design = NAME_SAI_SUPREMACY_ARKSHIP_TIER_3 }
                }
                set_graphical_culture = military_arkship
            }
            if = {
                limit = {
                    NOT = { has_modifier = sai_supremacy_arkship_stats }
                }
                add_modifier = { modifier = sai_supremacy_arkship_stats days = -1 }
            }
            if = {
                limit = { has_ship_flag = foc_v177_shell_restore_stats }
                remove_ship_flag = foc_v177_shell_restore_stats
            }
        }
    }
}

# Country scope. The owner difficulty synchronization can invoke this directly.
foc_v177_sync_supremacy_shell_country = {
    if = { limit = { is_ai = no } every_owned_ship = { foc_v177_sync_supremacy_shell_ship = yes } }
}
