# Planet scope: initialize every pre-authored Star Wars member colony. This
# follows the supplied Kuat origin's light-weight population pattern instead of
# generating a full homeworld population on all thirty-one planets. Coruscant
# receives the player's original starting population; every other member world
# receives one small pop group of the player's main species.
foc_republic_origin_prepare_member_world = {
    # V231o: never re-run vanilla colony generation on an existing story colony.
    # Re-entry may repair ownership/flags, but it must preserve all existing pops.
    if = {
        limit = { is_colony = no }
        create_colony = { owner = event_target:foc_republic_origin_republic }
        generate_start_deposits_and_blockers = yes
        clear_blockers = yes
        set_planet_flag = foc_republic_origin_member_world
        if = {
            limit = { NOT = { has_planet_flag = foc_sw_coruscant } }
            create_pop_group = {
                size = 100
                species = event_target:foc_republic_origin_republic.owner_main_species
            }
        }
    }
    else = {
        set_owner = event_target:foc_republic_origin_republic
        set_controller = event_target:foc_republic_origin_republic
        set_planet_flag = foc_republic_origin_member_world
    }
    foc_sw_restore_canonical_colony_name = yes
}

# Planet scope: replace the generated building set with one guaranteed capital
# building that carries the colony's administration, stability and direct
# economy. Districts remain generated normally, so the player can still develop
# every world while the nexus prevents unemployment-driven collapse.
foc_republic_origin_install_member_world_nexus = {
    # V231o: full economic initialization is one-shot. Older repair/map-fix
    # callers may invoke this effect again, but they may only restore the Nexus;
    # they can never wipe factories, droids or later story buildings again.
    if = {
        limit = { NOT = { has_carrier_flag = foc_v231o_member_world_layout_initialized } }
        inline_script = "foc_v232i_safe_colony_layout"
        remove_all_buildings = yes
        add_building = { building = building_foc_republic_origin_member_world_nexus }
        set_carrier_flag = foc_v231o_member_world_layout_initialized
    }
    else_if = {
        limit = { NOT = { has_building = building_foc_republic_origin_member_world_nexus } }
        if = {
            limit = { free_building_slots > 0 }
            add_building = { building = building_foc_republic_origin_member_world_nexus }
        }
    }
    foc_sw_restore_canonical_colony_name = yes
}

# Country scope (the player Republic): one shared entry point for every
# Separatist-surrender path. Narrative completion and recorded native war results
# use this guarded offer. V232s debug commits directly instead;
# both routes share the original separated post-war cleanup events.
foc_republic_origin_force_separatist_surrender = {
    if = {
        limit = {
            foc_v197_republic_story_actor = yes
            has_country_flag = foc_republic_origin_republic_country
            has_global_flag = foc_republic_origin_separatist_war_active
            NOT = { has_country_flag = foc_republic_origin_separatist_war_won }
            NOT = { has_country_flag = foc_republic_origin_separatist_surrender_staging_142 }

            # V232u: either story victory or confirmed engine war result; not a debug bypass.
            NOT = { has_country_flag = foc_v232o_separatist_surrender_offer_pending }
            foc_v232o_separatist_victory_condition = yes
        }
        set_country_flag = foc_republic_origin_separatist_capital_fallen
        set_country_flag = foc_v232o_separatist_surrender_offer_pending
        if = {
            limit = { foc_v232s_cis_plot_victory = yes }
            log = "FOC_V232S_SURRENDER: story route satisfied; surrender offered."
        }
        else = {
            log = "FOC_V232U_SURRENDER: verified native Clone Wars victory; post-war surrender offered without a plot-score requirement."
        }
        country_event = { id = foc_republic_origin.20 }
    }
}

    }

    # later hosts Death Star II, so make the assignment explicit even if a map
    # compatibility file changes Endor's peacetime regional flag.
    if = {

            exists = event_target:foc_republic_origin_empire
            exists = event_target:foc_swmap_endor_system

            foc_republic_origin_claim_for_empire = yes

        foc_republic_origin_claim_for_rebels = yes
    }
    event_target:foc_swmap_hoth_system = {
        foc_republic_origin_claim_for_rebels = yes
    }

    every_system = {

                has_star_flag = foc_swmap_ferrix_home
                has_star_flag = foc_swmap_onderon_home

                has_star_flag = foc_v175_atollon_home

    event_target:foc_khan_rebel_system = {
        foc_republic_origin_claim_for_rebels = yes
    }
    event_target:foc_khan_rebel_alliance = {

        set_country_flag = foc_republic_origin_imperial_rebel_country

        # hulls. They do not belong in the Imperial Rebellion; the class stays

        foc_republic_origin_remove_rebel_nebula_ships = yes

        foc_republic_origin_apply_rebellion_war_effort_by_difficulty = yes

# Country scope (Imperial Rebels): remove every event-spawned Nebula-class
# vessel inherited from the reusable Rebel-crisis fleet package. The monthly
# story pulse also calls this so reinforcements and old saves cannot reintroduce
# the class before the New Republic is founded.
# 本补丁修复：这个清理一共有五个调用点，其中 foc_republic_origin.140 的那一处
# 是**每月执行**的。原意是「叛军阶段不该出现星云级」，但它只看船体尺寸，不看这个
# 国家到底有没有星云级的建造权——于是玩家合法造出来的星云级下个月就被自己人删掉。
#
# v16 我曾在调用点上加过一个全局旗标判定，但那个旗标只有「双死星 + 夸特」那条
# 授予路径会置。通过银河条约或手动补发拿到星云级的国家旗标是空的，清理照跑。
#
# 现在把判定移进效果本身，一处覆盖全部五个调用点，并且换成语义正确的条件：
# **持有星云级蓝图科技的国家，它的星云级永远不删。** 没有蓝图的（叛军阶段、
# 危机初始化器误生成的那些）行为与原来完全一致。
foc_republic_origin_remove_rebel_nebula_ships = {
    if = {
        limit = { NOT = { has_technology = tech_foc_rebel_nebula_blueprint } }
        every_owned_fleet = {
            every_owned_ship = {
                limit = { OR = { is_ship_size = foc_rebel_nebula is_ship_size = foc_rebel_nebula_foc_lowest } }
                delete_ship = this
            }
        }
    }
}

# Country scope (the Republic): give the union an actual navy infrastructure.
# Claiming twenty-seven systems with outposts was the right call for the day-one
# budget, but it left the greatest power in the galaxy with no starport, no
# shipyard and no defended system anywhere. Three worlds are upgraded instead:
# the capital, the shipyard world, and Rendili's naval works. The remaining
# twenty-four keep their outposts, so the deficit that citadels everywhere used
# to cause does not come back.
foc_republic_origin_establish_core_starbases = {
    # Coruscant: the seat of government and the Home Fleet's anchorage.
    if = {
        limit = { exists = event_target:foc_sw_coruscant_system }
        event_target:foc_sw_coruscant_system = {
            if = {
                limit = { NOT = { exists = starbase } }
                create_starbase = {
                    size = starbase_outpost
                    owner = event_target:foc_republic_origin_republic
                }
            }
            starbase = {
                set_owner = event_target:foc_republic_origin_republic
                set_starbase_size = starbase_citadel
                set_starbase_module = { slot = 1 module = shipyard }
                set_starbase_module = { slot = 2 module = shipyard }
                set_starbase_module = { slot = 3 module = anchorage }
                set_starbase_module = { slot = 4 module = gun_battery }
                set_starbase_building = { slot = 1 building = crew_quarters }
                set_ship_construction_type = starbase_shipyard
                set_starbase_flag = foc_republic_origin_core_starbase
            }
        }
    }

    # Kuat: the yards themselves. The Mega Shipyard is already in orbit; it
    # should not be the only place in the Republic that can lay a keel.
    if = {
        limit = { exists = event_target:foc_sw_kuat_system }
        event_target:foc_sw_kuat_system = {
            if = {
                limit = { NOT = { exists = starbase } }
                create_starbase = {
                    size = starbase_outpost
                    owner = event_target:foc_republic_origin_republic
                }
            }
            starbase = {
                set_owner = event_target:foc_republic_origin_republic
                set_starbase_size = starbase_starhold
                set_starbase_module = { slot = 1 module = shipyard }
                set_starbase_module = { slot = 2 module = shipyard }
                set_starbase_module = { slot = 3 module = shipyard }
                set_starbase_building = { slot = 1 building = crew_quarters }
                set_ship_construction_type = starbase_shipyard
                set_starbase_flag = foc_republic_origin_core_starbase
            }
        }
    }

    # Rendili: the Republic's other great naval works, and the Core's picket.
    if = {
        limit = {
            any_system = { has_star_flag = foc_sw_rendili_home }
        }
        random_system = {
            limit = { has_star_flag = foc_sw_rendili_home }
            if = {
                limit = { NOT = { exists = starbase } }
                create_starbase = {
                    size = starbase_outpost
                    owner = event_target:foc_republic_origin_republic
                }
            }
            starbase = {
                set_owner = event_target:foc_republic_origin_republic
                set_starbase_size = starbase_starport
                set_starbase_module = { slot = 1 module = shipyard }
                set_starbase_module = { slot = 2 module = anchorage }
                set_ship_construction_type = starbase_shipyard
                set_starbase_flag = foc_republic_origin_core_starbase
            }
        }
    }
}

# Any scope: guarantee the Imperial civil war is actually being fought. The
# declaration in foc_republic_origin_start_imperial_rebellion is issued in the
# same tick as the transformation, the annexation and the fleet spawns, and a
# declaration made during that much ownership churn does not always take. This
# runs monthly, so the war starts even if the opening declaration was refused,
# and it restarts if the two sides ever end up at peace while the civil war is
# still supposed to be running.
foc_republic_origin_enforce_civil_war = {
    # V182 keeps the ordinary routes one-shot. V197 adds a narrow ten-day-lock
    # retry only for the all-AI Hutt observer route, where same-tick ownership
    # churn can otherwise leave two hostile countries without a war object.
    if = {
        limit = {
            has_global_flag = foc_republic_origin_empire_war_active
            exists = event_target:foc_republic_origin_empire
            exists = event_target:foc_khan_rebel_alliance
        }
        event_target:foc_republic_origin_empire = {
            establish_communications_no_message = event_target:foc_khan_rebel_alliance
            set_faction_hostility = {
                target = event_target:foc_khan_rebel_alliance
                set_hostile = yes
                set_neutral = no
                set_friendly = no
            }
            if = {
                limit = {
                    has_country_flag = foc_v197_hutt_era_ai_empire_closer
                    NOT = { is_at_war_with = event_target:foc_khan_rebel_alliance }
                    NOT = { has_country_flag = foc_republic_origin_civil_war_declaration_lock }
                }
                set_timed_country_flag = {
                    flag = foc_republic_origin_civil_war_declaration_lock
                    days = 10
                }
                declare_war = {
                    target = event_target:foc_khan_rebel_alliance
                    attacker_war_goal = wg_foc_republic_civil_war
                }
            }
        }
        event_target:foc_khan_rebel_alliance = {
            establish_communications_no_message = event_target:foc_republic_origin_empire
            set_faction_hostility = {
                target = event_target:foc_republic_origin_empire
                set_hostile = yes
                set_neutral = no
                set_friendly = no
            }
        }
    }
}

# Any scope. One-time deployment per Rebel-held system. A destroyed or moved
# garrison is never automatically replaced; conquest reinforcements are separate.
# 本补丁改动：原实现挂在月度剧情脉冲上，只要某个义军星系当前没有驻军舰队
# 就补一支，于是被打掉的驻军每个月都会原地复活，帝国的战果无法累积。
#
# 现在改成一次性布防：每个星系用星系旗标 foc_republic_origin_rebel_garrisoned
# 记录「已经布过防」，此后无论驻军是否阵亡都不再补。义军的兵力增长改由
# foc_saga_rebel_conquest_reinforcement 负责——打下一个星系，发一组舰队。
foc_republic_origin_garrison_rebel_systems = {
    if = {
        limit = { exists = event_target:foc_khan_rebel_alliance }
        every_system = {
            limit = {
                exists = space_owner
                space_owner = { is_same_value = event_target:foc_khan_rebel_alliance }
                NOT = { has_star_flag = foc_republic_origin_rebel_garrisoned }
                NOT = {
                    any_fleet_in_system = {
                        exists = owner
                        owner = { is_same_value = event_target:foc_khan_rebel_alliance }
                        has_fleet_flag = foc_republic_origin_rebellion_garrison
                    }
                }
            }
            set_star_flag = foc_republic_origin_rebel_garrisoned
            foc_republic_origin_spawn_rebellion_garrison_fleet = yes
        }
    }
}

# ---------------------------------------------------------------------------
# V32 Imperial fear deficit. Applied to the Galactic Empire for as long as the
# New Order has no answer to the question event .181 asks. Cleared only when
# the Death Star is actually built AND the Tarkin Doctrine has been adopted.
# ---------------------------------------------------------------------------
foc_republic_origin_update_imperial_fear_deficit = {
    every_country = {
        limit = { has_country_flag = foc_republic_origin_empire_country }
        if = {
            limit = {
                has_country_flag = foc_v109_jedha_fear_order_established
                NOT = { has_country_flag = foc_republic_origin_death_star_two_destroyed }
            }
            if = {
                limit = { has_modifier = foc_republic_origin_imperial_fear_deficit }
                remove_modifier = foc_republic_origin_imperial_fear_deficit
            }
        }
        else = {
            if = {
                limit = { NOT = { has_modifier = foc_republic_origin_imperial_fear_deficit } }
                add_modifier = { modifier = foc_republic_origin_imperial_fear_deficit days = -1 }
            }
        }
    }
}
