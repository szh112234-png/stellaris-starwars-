# V79 - actual game effects for canonical story windows

foc_canon_create_ahsoka = {
    if = {
        limit = { NOT = { exists = event_target:foc_canon_ahsoka } NOT = { any_owned_leader = { has_leader_flag = foc_canon_ahsoka_leader } } }
        create_leader = {
            tier = leader_tier_legendary
            class = commander
            species = owner_main_species
            name = "FOC_CANON_AHSOKA_NAME"
            gender = female
            skill = 6
            event_leader = yes
            skip_background_generation = yes
            randomize_traits = no
            effect = {
                change_leader_portrait = foc_saga_ahsoka_01
                set_leader_flag = foc_canon_ahsoka_leader
                add_trait = { trait = leader_trait_foc_force_sensitive consume_selection = no }
                add_trait = { trait = leader_trait_psionic consume_selection = no }
                save_global_event_target_as = foc_canon_ahsoka
            }
        }
    }
}

foc_canon_remove_ahsoka = {
    every_owned_leader = {
        limit = { has_leader_flag = foc_canon_ahsoka_leader }
        kill_leader = { show_notification = no }
    }
    if = {
        limit = { exists = event_target:foc_canon_ahsoka }
        clear_global_event_target = foc_canon_ahsoka
    }
}

foc_canon_starkiller_construction_effect = {
    set_global_flag = foc_canon_starkiller_construction_started
    if = {
        limit = { has_origin = origin_fop_first_order }
        # Opening First Order project: reward and burden apply only to the player.
        # Do not alter the New Republic or Imperial Remnant Fallen Empires.
        add_modifier = { modifier = foc_canon_starkiller_industry days = 3600 }
        add_resource = { alloys = 25000 engineering_research = 5000 }
        add_modifier = { modifier = fop_origin_starkiller_project_burden days = -1 }
    }
    else = {
        every_country = {
            limit = {
                OR = {
                    has_country_flag = foc_sw_imperial_remnant_country
                    has_country_flag = foc_khan_first_order_country
                    has_country_flag = foc_republic_origin_became_first_order
                }
            }
            add_modifier = { modifier = foc_canon_starkiller_industry days = 3600 }
            add_resource = { alloys = 25000 engineering_research = 5000 }
        }
    }
}

# V85: first completed battle of each historical phase. These are parallel anchors;
# they never replace the existing era anchors (Clone War, Order 66, Treaty, Phase I...).
foc_canon_mark_clone_war_first_battle = {
    if = {
        limit = { NOT = { has_country_flag = foc_canon_clone_war_first_battle_anchor } }
        set_country_flag = foc_canon_clone_war_first_battle_anchor
        set_global_flag = foc_canon_clone_war_first_battle_recorded
        # Generic Clone-War narrative beats now count from the first completed battle.
        set_timed_country_flag = { flag = foc_canon_wait_mortis days = 360 }
        set_timed_country_flag = { flag = foc_canon_wait_tup_inquisitors days = 720 }
        set_timed_country_flag = { flag = foc_canon_wait_ahsoka_leaves days = 900 }
        set_timed_country_flag = { flag = foc_canon_wait_yoda_ghost days = 1080 }
    }
}
