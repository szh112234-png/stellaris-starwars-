foc_ai_capital_register_mc80_design = {
    if = {
        limit = {
            is_ai = yes
            foc_ai_capital_is_mc80_country = yes
        }
        if = {
            limit = { NOT = { has_country_flag = foc_rebel_mc80_design_unlocked } }
            set_country_flag = foc_rebel_mc80_design_unlocked
            set_global_flag = foc_rebel_mc80_design_first_unlocked
        }
        if = {
            limit = { NOT = { has_country_flag = foc_rebel_mc80_design_registered_v131 } }
            foc_story_grant_imperial_rebel_mc80_design = yes
        }
    }
}

# The scripted Fallen-Empire New Republic needs the special designs; ordinary
# Treaty successors need the non-special _NR designs. Register both through the
# existing guarded effects, then let country type decide which channel is shown.
foc_ai_capital_register_new_republic_designs = {
    if = {
        limit = {
            foc_ai_capital_is_new_republic_country = yes
            NOT = { has_country_flag = foc_ai_capital_nr_designs_registered_v1 }
        }
        # Set the construction authorization before creating fixed designs;
        # ship-size visibility is identity-gated, while the final build button
        # remains locked behind this flag.
        set_country_flag = foc_saga_new_republic_shipyard_access
        foc_republic_origin_repair_new_republic_ship_blueprints = yes
        if = {
            limit = { has_global_flag = foc_republic_origin_galactic_treaty }
            foc_saga_open_new_republic_shipyards = yes
        }
        set_country_flag = foc_ai_capital_nr_designs_registered_v1
        refresh_auto_generated_ship_designs = yes
    }
}
