# FOC V158 - Galactic Treaty sequel-era clock and Imperial Remnant asset repair.
# All Stellaris years use 360 days.

foc_v158_start_treaty_sequel_timeline = {
    if = {
        limit = { NOT = { has_global_flag = foc_v158_treaty_sequel_timeline_started } }
        set_global_flag = foc_v158_treaty_sequel_timeline_started
        # Treaty +4y: Luke rebuilds the Jedi Temple on Ossus.
        set_timed_global_flag = { flag = foc_v158_treaty_luke_4y_delay days = 1440 }
        # Treaty +10y: Ben Solo begins training under Luke (narrative only).
        set_timed_global_flag = { flag = foc_v158_treaty_ben_10y_delay days = 3600 }
        # Treaty +23y: Ben destroys the temple; Leia leaves the New Republic.
        set_timed_global_flag = { flag = foc_v158_treaty_temple_23y_delay days = 8280 }
        log = "FOC_V158_TIMELINE: Treaty sequel clock started: Luke +4y, Ben training +10y, Temple/Leia +23y, War in Heaven +24y."
    }
}

# Country scope: Treaty Imperial Remnant. Restore the original scripted FE
# infrastructure before the War in Heaven, and repair V157 saves where the
# Executor fixed design was not registered before the eight groups spawned.
foc_v158_restore_imperial_remnant_treaty_assets = {
    save_global_event_target_as = foc_sw_imperial_remnant
    if = {
        limit = { NOT = { has_modifier = foc_v158_imperial_remnant_starbase_capacity } }
        add_modifier = { modifier = foc_v158_imperial_remnant_starbase_capacity days = -1 }
    }

    # Keep the sealed design registered, but V159 no longer replenishes
    # Executor combat losses from this monthly-compatible infrastructure repair.
    foc_story_grant_jakku_executor_blueprint = yes

    # Before the War in Heaven every canonical Remnant system is restored to the
    # same Citadel standard used by the original Fallen-Empire setup.
    if = {
        limit = { NOT = { has_global_flag = foc_sw_war_in_heaven_started } }
        every_system = {
            limit = {
                has_star_flag = foc_sw_imperial_remnant_cluster
                OR = {
                    NOT = { exists = space_owner }
                    space_owner = { is_same_value = root }
                }
            }
            foc_sw_claim_imperial_remnant_system = yes
        }
    }
}
