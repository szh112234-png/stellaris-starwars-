foc_lowest_ds_sync_ship = {
    if = {
        limit = { foc_lowest_ds_is_managed_ship = yes is_in_combat = no }
        if = {
            limit = { has_global_flag = foc_khan_strength_lowest }
            if = {
                limit = { has_ship_flag = foc_republic_origin_death_star_two }
                if = {
                    limit = { OR = { foc_lowest_ds_is_mobile_two = yes has_ship_flag = foc_ds2_mobile_completed } }
                    if = {
                        limit = { NOT = { has_component = FOC_LOWEST_DS_II_FOC_DEATH_STAR_II_CORE_COMPONENT } }
                        foc_lowest_ds_apply_completed_two = yes
                    }
                }
                else_if = {
                    limit = { NOT = { has_component = FOC_LOWEST_DS_II_UNFINISHED_FOC_DEATH_STAR_II_CORE_COMPONENT } }
                    foc_lowest_ds_apply_unfinished_two = yes
                }
            }
            else_if = {
                limit = { has_ship_flag = foc_republic_origin_death_star_one NOT = { has_component = FOC_LOWEST_DS_I_DEATH_STAR_CORE_COMPONENT } }
                set_ship_design = { design = NAME_FOC_REPUBLIC_ORIGIN_DEATH_STAR_FOC_LOWEST }
            }
            if = {
                limit = {
                    NOT = { foc_lowest_ds_is_story_ship = yes }
                    OR = { is_ship_size = deathstarbattlestation has_ship_flag = foc_lowest_ds_generic_origin }
                }
                if = {
                    limit = { NOT = { has_component = FOC_LOWEST_DS_I_DEATH_STAR_CORE_COMPONENT } }
                    set_ship_design = { design = NAME_FOC_REPUBLIC_ORIGIN_DEATH_STAR_FOC_LOWEST }
                }
                set_ship_flag = foc_lowest_ds_generic_origin
            }
            set_ship_flag = foc_lowest_ds_mode_applied
        }
        else_if = {
            limit = {
                OR = {
                    has_ship_flag = foc_lowest_ds_mode_applied
                    has_component = FOC_LOWEST_DS_I_DEATH_STAR_CORE_COMPONENT
                    has_component = FOC_LOWEST_DS_II_UNFINISHED_FOC_DEATH_STAR_II_CORE_COMPONENT
                    has_component = FOC_LOWEST_DS_II_FOC_DEATH_STAR_II_CORE_COMPONENT
                }
            }
            if = {
                limit = { has_ship_flag = foc_republic_origin_death_star_two }
                if = { limit = { OR = { foc_lowest_ds_is_mobile_two = yes has_ship_flag = foc_ds2_mobile_completed } } foc_lowest_ds_apply_completed_two = yes }
                else = { foc_lowest_ds_apply_unfinished_two = yes }
            }
            else = { set_ship_design = { design = NAME_FOC_REPUBLIC_ORIGIN_DEATH_STAR } }
            remove_ship_flag = foc_lowest_ds_mode_applied
        }
    }
}
