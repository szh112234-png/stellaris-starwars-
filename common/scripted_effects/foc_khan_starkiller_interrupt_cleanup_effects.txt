# 1.78 crisis-only Starkiller cleanup.
# This effect must never touch the player's myk_starkiller_target_system,
# myk_starkiller_base_megastructure, or player busy/cooldown flags.
# It is called when the First Order Starkiller capital/base is destroyed.
foc_khan_cleanup_interrupted_starkiller_shot = {
    # Remove exact crisis origin markers saved by event 8200.
    if = {
        limit = { exists = event_target:myk_starkiller_crisis_fire_origin_location }
        event_target:myk_starkiller_crisis_fire_origin_location = { remove_megastructure = this }
    }
    if = {
        limit = { exists = event_target:myk_starkiller_crisis_beacon_origin_location }
        event_target:myk_starkiller_crisis_beacon_origin_location = { remove_megastructure = this }
    }

    # Remove only the crisis base and origin-side objects in Ilum. Target-side
    # marker types are intentionally excluded because a player may be firing at
    # the First Order capital at the same moment.
    random_system = {
        limit = { has_star_flag = foc_khan_ilum_home }
        every_system_megastructure = {
            limit = {
                OR = {
                    has_megastructure_flag = foc_khan_crisis_starkiller
                    is_megastructure_type = myk_starkiller_base_final
                    is_megastructure_type = myk_starkiller_sun_drain
                    is_megastructure_type = myk_starkiller_fire_origin_marker
                    is_megastructure_type = myk_starkiller_beacon_origin_marker
                }
            }
            remove_megastructure = this
        }
    }

    # A committed crisis shot remains unstoppable. Preserve its target system,
    # pending flag, and target markers until event 8201 resolves it exactly once.
    if = {
        limit = { NOT = { has_global_flag = foc_khan_forced_starkiller_in_progress } }

        if = {
            limit = { exists = event_target:myk_starkiller_crisis_fire_target_location }
            event_target:myk_starkiller_crisis_fire_target_location = { remove_megastructure = this }
        }
        if = {
            limit = { exists = event_target:myk_starkiller_crisis_beacon_target_location }
            event_target:myk_starkiller_crisis_beacon_target_location = { remove_megastructure = this }
        }
        if = {
            limit = { exists = event_target:myk_starkiller_crisis_target_system }
            event_target:myk_starkiller_crisis_target_system = {
                remove_star_flag = myk_starkiller_crisis_shot_pending
            }
        }

        if = { limit = { exists = event_target:myk_starkiller_crisis_target_system } clear_global_event_target = myk_starkiller_crisis_target_system }
        if = { limit = { exists = event_target:myk_starkiller_crisis_target_star } clear_global_event_target = myk_starkiller_crisis_target_star }
        if = { limit = { exists = event_target:myk_starkiller_crisis_fire_target_location } clear_global_event_target = myk_starkiller_crisis_fire_target_location }
        if = { limit = { exists = event_target:myk_starkiller_crisis_beacon_target_location } clear_global_event_target = myk_starkiller_crisis_beacon_target_location }
    }

    if = { limit = { exists = event_target:myk_starkiller_crisis_base_megastructure } clear_global_event_target = myk_starkiller_crisis_base_megastructure }
    if = { limit = { exists = event_target:myk_starkiller_crisis_base_owner } clear_global_event_target = myk_starkiller_crisis_base_owner }
    if = { limit = { exists = event_target:myk_starkiller_crisis_base_system } clear_global_event_target = myk_starkiller_crisis_base_system }
    if = { limit = { exists = event_target:myk_starkiller_crisis_base_planet } clear_global_event_target = myk_starkiller_crisis_base_planet }
    if = { limit = { exists = event_target:myk_starkiller_crisis_fire_origin_location } clear_global_event_target = myk_starkiller_crisis_fire_origin_location }
    if = { limit = { exists = event_target:myk_starkiller_crisis_beacon_origin_location } clear_global_event_target = myk_starkiller_crisis_beacon_origin_location }
}
