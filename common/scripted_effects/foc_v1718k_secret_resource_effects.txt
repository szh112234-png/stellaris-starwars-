# Country scope. Updates only this feature's display marker/end state.
# Called on start/load/monthly; this effect NEVER credits any resources.
foc_v1718k_sync_secret_resource_collection = {
    if = {
        limit = { has_origin = origin_fop_first_order }
        if = {
            limit = {
                OR = {
                    has_country_flag = foc_v1718k_secret_resource_collection_ended
                    has_country_flag = foc_v1717_starkiller_was_completed
                    foc_v1717_starkiller_completed_for_arsenal = yes
                }
            }
            # Historical completion is terminal, even if the base is later lost.
            set_country_flag = foc_v1718k_secret_resource_collection_ended
            if = {
                limit = { has_modifier = foc_v1718k_secret_resource_collection }
                remove_modifier = foc_v1718k_secret_resource_collection
            }
        }
        else_if = {
            limit = { NOT = { has_modifier = foc_v1718k_secret_resource_collection } }
            add_modifier = { modifier = foc_v1718k_secret_resource_collection days = -1 }
        }
    }
    else_if = {
        limit = { has_modifier = foc_v1718k_secret_resource_collection }
        remove_modifier = foc_v1718k_secret_resource_collection
    }
}
