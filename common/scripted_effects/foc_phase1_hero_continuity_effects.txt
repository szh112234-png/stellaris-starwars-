# FOC Phase I - Historical Character Continuity
# Story leaders move between political eras instead of being re-created as duplicates.

foc_phase1_move_ahsoka_here = {
    if = {
        limit = { NOT = { any_owned_leader = { has_leader_flag = foc_canon_ahsoka_leader } } }
        random_country = {
            limit = { NOT = { is_same_value = root } any_owned_leader = { has_leader_flag = foc_canon_ahsoka_leader } }
            random_owned_leader = {
                limit = { has_leader_flag = foc_canon_ahsoka_leader }
                save_event_target_as = foc_phase1_moving_ahsoka
                root = { clone_leader = { target = event_target:foc_phase1_moving_ahsoka skip_background_generation = yes tier = leader_tier_legendary } }
                kill_leader = { show_notification = no }
            }
        }
    }
}
