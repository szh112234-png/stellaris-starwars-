# FOC 3.11v staged Resistance-system helpers.

# Scope: a newly spawned Resistance system.
foc_khan_claim_rebel_stage_system_311v = {
    # V235b: destroyed/lost bases are never initialization failures.
    if = {
        limit = { NOT = { has_star_flag = star_cracked } }
        if = {
            limit = { exists = starbase }
            starbase = {
                set_owner = event_target:foc_khan_rebel_alliance
                set_starbase_size = starbase_starhold
                set_starbase_flag = foc_khan_rebel_stage_base_311v
            }
        }
        else = {
            create_starbase = {
                size = starbase_starhold
                owner = event_target:foc_khan_rebel_alliance
                effect = { set_starbase_flag = foc_khan_rebel_stage_base_311v }
            }
        }
    }
}

# Scope: a newly spawned habitable Resistance world.
foc_khan_prepare_rebel_stage_colony_311v = {
    # V235b: destroyed/lost bases are never initialization failures.
    if = {
        limit = { foc_v235b_base_planet_usable = yes NOT = { is_colony = yes } }
        create_colony = { owner = event_target:foc_khan_rebel_alliance }
        generate_start_deposits_and_blockers = yes
        clear_blockers = yes
        colony = {
            inline_script = "foc_v232i_safe_colony_layout"
            generate_start_pops = yes
            generate_home_planet_final_pass = yes
        }
        while = {
            count = 8
            create_army = {
                owner = event_target:foc_khan_rebel_alliance
                species = event_target:foc_khan_rebel_species
                type = defense_army
            }
        }
    }
}
