# FOC 3.07 - shared setup, fleet, defense and sustainable economy effects for the two scripted Fallen Empires.

# Replaces generated buildings with a direct-output ancient infrastructure suite.
# Scope: an owned colony planet.
foc_sw_apply_fallen_empire_buildings = {
    set_carrier_flag = fallen_empire_world

    # V231o: this effect is called by initial setup, Treaty conversion and
    # recurring FE repair. Only the first call may replace the generated economy.
    # Later calls repair the required primary building without touching pops or
    # unrelated story buildings.
    if = {
        limit = { NOT = { has_carrier_flag = foc_v231o_fe_layout_initialized } }
        remove_all_buildings = yes

        if = {
            limit = {
                exists = owner
                OR = {
                    has_carrier_flag = foc_sw_hosnian_prime
                    has_carrier_flag = foc_sw_bastion
                }
                owner = {
                    OR = {
                        has_country_flag = foc_sw_new_republic_country
                        has_country_flag = foc_sw_imperial_remnant_country
                    }
                }
            }
            owner = {
                if = {
                    limit = { NOT = { has_technology = tech_powered_exoskeletons } }
                    give_technology = { tech = tech_powered_exoskeletons message = no }
                }
            }
            set_timed_carrier_flag = { flag = ignore_ai_building_limitations days = 2 }
            # V232n: when robotics is already known, occupy the reserved primary slot
            # before adding the five support buildings. Otherwise the original later
            # species-first seed pass grants robotics and calls this installer.
            foc_v232m_ensure_vanilla_droid_factory = yes
        }
        else = {
            if = { limit = { free_building_slots > 0 } add_building = { building = building_foc_sw_fe_civil_nexus } }
        }
        if = { limit = { free_building_slots > 0 } add_building = { building = building_foc_sw_fe_power_core } }
        if = { limit = { free_building_slots > 0 } add_building = { building = building_foc_sw_fe_matter_fabricator } }
        if = { limit = { free_building_slots > 0 } add_building = { building = building_foc_sw_fe_matter_fabricator } }
        if = { limit = { free_building_slots > 0 } add_building = { building = building_foc_sw_fe_ship_foundry } }
        if = { limit = { free_building_slots > 0 } add_building = { building = building_foc_sw_fe_support_complex } }
        set_carrier_flag = foc_v231o_fe_layout_initialized
    }
    else = {
        # Capital rule: Hosnian Prime/Bastion use the vanilla Robot Assembly
        # Plant instead of Ancient Civil Nexus. If an earlier pass left a Nexus
        # occupying the last slot, remove only that building and replace it.
        if = {
            limit = {
                exists = owner
                OR = {
                    has_carrier_flag = foc_sw_hosnian_prime
                    has_carrier_flag = foc_sw_bastion
                    is_capital = yes
                }
                owner = {
                    OR = {
                        has_country_flag = foc_sw_new_republic_country
                        has_country_flag = foc_sw_imperial_remnant_country
                    }
                }
                NOR = { has_building = building_robot_assembly_plant has_building = building_robot_assembly_complex }
            }
            owner = {
                if = {
                    limit = { NOT = { has_technology = tech_powered_exoskeletons } }
                    give_technology = { tech = tech_powered_exoskeletons message = no }
                }
            }
            set_timed_carrier_flag = { flag = ignore_ai_building_limitations days = 2 }
            foc_v232m_ensure_vanilla_droid_factory = yes
        }
        else_if = {
            limit = {
                NOT = {
                    OR = {
                        has_carrier_flag = foc_sw_hosnian_prime
                        has_carrier_flag = foc_sw_bastion
                        is_capital = yes
                    }
                }
                NOR = {
                    has_building = building_foc_sw_fe_civil_nexus
                    has_building = building_robot_assembly_plant
                    has_building = building_robot_assembly_complex
                }
                free_building_slots > 0
            }
            add_building = { building = building_foc_sw_fe_civil_nexus }
        }
    }
}

foc_sw_claim_new_republic_system = {
    if = {
        limit = {
            NOT = { has_star_flag = foc_v196_hutt_criminal_empire_system }
            # V204: this older FE claim helper bypasses the Republic-origin
            # claim effect, so it needs its own D'Qar/Resistance lock.
            NOT = {
                AND = {
                    has_star_flag = foc_khan_rebel_dqar_home
                    OR = {
                        has_global_flag = foc_v161_dqar_resistance_founded
                        AND = {
                            OR = {
                                has_global_flag = foc_v158_leia_resistance_foundation_due
                                has_global_flag = foc_v109_bloodline_resistance_authorized
                            }
                            any_country = {
                                has_country_flag = foc_khan_rebel_alliance_country
                                OR = {
                                    has_country_flag = foc_khan_first_order_resistance_country
                                    has_country_flag = foc_republic_origin_post_treaty_rebel_country
                                    has_country_flag = foc_republic_origin_dqar_inheritance_complete
                                }
                            }
                        }
                    }
                }
            }
        }
    if = {
        limit = { exists = starbase }
        starbase = {
            set_owner = event_target:foc_sw_new_republic
            set_starbase_size = starbase_citadel
            set_starbase_flag = foc_sw_new_republic_starbase
        }
    }
    else = {
        create_starbase = {
            size = starbase_citadel
            owner = event_target:foc_sw_new_republic
            effect = { set_starbase_flag = foc_sw_new_republic_starbase }
        }
    }
    }
}

# Starbase scope. Finish an untouched scripted ring once. Existing player/AI
# module choices are preserved, and repeated setup or succession calls do nothing.
# Checking actual size/module counts first also avoids rebuilding a correct old-save
# ring merely because it predates the initialization marker.
foc_starbase_finish_scripted_shipyard_ring = {
    if = {
        limit = { NOT = { has_starbase_flag = foc_starbase_shipyard_ring_initialized } }
        if = {
            limit = {
                OR = {
                    count_starbase_modules = { include_being_constructed = yes count = 0 }
                    AND = {
                        count_starbase_modules = { include_being_constructed = yes count = 1 }
                        count_starbase_modules = { type = orbital_ring_shipyard include_being_constructed = yes count = 1 }
                    }
                    AND = {
                        count_starbase_modules = { include_being_constructed = yes count = 2 }
                        count_starbase_modules = { type = orbital_ring_shipyard include_being_constructed = yes count = 2 }
                    }
                    AND = {
                        count_starbase_modules = { include_being_constructed = yes count = 3 }
                        count_starbase_modules = { type = orbital_ring_shipyard include_being_constructed = yes count = 3 }
                    }
                    AND = {
                        count_starbase_modules = { include_being_constructed = yes count = 4 }
                        count_starbase_modules = { type = orbital_ring_shipyard include_being_constructed = yes count = 4 }
                    }
                }
            }
            if = {
                limit = { NOT = { has_starbase_size = orbital_ring_tier_3 } }
                set_starbase_size = orbital_ring_tier_3
            }
            if = {
                limit = { count_starbase_modules = { type = orbital_ring_shipyard include_being_constructed = yes count < 4 } }
                set_starbase_module = { slot = 1 module = orbital_ring_shipyard }
                set_starbase_module = { slot = 2 module = orbital_ring_shipyard }
                set_starbase_module = { slot = 3 module = orbital_ring_shipyard }
                set_starbase_module = { slot = 4 module = orbital_ring_shipyard }
            }
            set_ship_construction_type = starbase_shipyard
        }
        set_starbase_flag = foc_starbase_shipyard_ring_initialized
    }
}
