# Original Arkship modules remain unchanged for normal Arkships but are hidden from the Supremacy.
# Supremacy-specific clones preserve all gameplay effects while using invisible section entities.

arkship_engineering_bay = {
	icon = "GFX_arkship_engineering_bay"
	section = "ARKSHIP_CONSTRUCTION_MODULE"
	construction_days = 180
	starbase_type = arkship
	category = sm_naval_support_cat

	potential = {
		owner? = {
			OR = {
				has_technology = tech_space_construction
				has_country_flag = champions_forge_enclave_country
			}
		}
		is_arkship_starbase = yes
		NOR = {
			OR = { has_starbase_size = sai_supremacy_arkship_tier_1 has_starbase_size = sai_supremacy_arkship_tier_1_shell_foc_lowest }
			OR = { has_starbase_size = sai_supremacy_arkship_tier_2 has_starbase_size = sai_supremacy_arkship_tier_2_shell_foc_lowest }
			OR = { OR = { has_starbase_size = sai_supremacy_arkship_tier_3 has_starbase_size = sai_supremacy_arkship_tier_3_foc_lowest } has_starbase_size = sai_supremacy_arkship_tier_3_shell_foc_lowest }
			# 本补丁修复：危机版至尊号走的是 foc_ 前缀的另一套尺寸
			# （foc_supremacy_crisis_events.txt 用 set_starbase_size 把
			# military_arkship_tier_N 改成 foc_sai_supremacy_arkship_tier_N），
			# 而这些模块的 potential 只认无前缀的三个尺寸，于是危机版一改尺寸，
			# 全部模块的 potential 立刻不成立——舰船上一个功能都没有。
			OR = { has_starbase_size = foc_sai_supremacy_arkship_tier_1 has_starbase_size = foc_sai_supremacy_arkship_tier_1_foc_lowest }
			OR = { has_starbase_size = foc_sai_supremacy_arkship_tier_2 has_starbase_size = foc_sai_supremacy_arkship_tier_2_foc_lowest }
			OR = { has_starbase_size = foc_sai_supremacy_arkship_tier_3 has_starbase_size = foc_sai_supremacy_arkship_tier_3_foc_lowest }
		}
	}

	possible = {
		count_starbase_modules = {
			type = arkship_beastport
			count < 1
			include_being_constructed = yes
		}
	}

	show_in_tech = "tech_space_construction"

	show_tech_unlock_if = {
		is_nomadic = yes
	}

	cannot_fit_on_slots = { spinal_mount_1 }

	resources = {
		category = starbase_arkship_upgrades
		cost = {
			alloys = @arkship_cost_alloys_100
		}

		upkeep = {
			energy = @arkship_upkeep_energy_2
		}
	}

	station_modifier = {
		show_only_custom_tooltip = no
		megastructure_build_speed_mult = 0.15
	}
	planet_modifier = {
		planet_building_capacity_add = 1
	}

	on_finished = {
		set_ship_construction_type = starbase_shipyard
	}
	on_destroyed = {
		if = {
			limit = { NOT = { has_starbase_module = arkship_engineering_bay } }
			set_ship_construction_type = none
		}
	}

	inline_script = {
		script = jobs/arkship_module_soldiers_add
		ZONE = zone_ark_shipwright
		AMOUNT = @arkship_module_jobs
	}

	ai_weight = {
		weight = 100 #use same weight here - ai weighting is done in starbase_types.txt
	}
}
