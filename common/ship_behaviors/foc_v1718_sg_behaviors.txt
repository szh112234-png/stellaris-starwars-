ship_behavior = {
	name = "foc_v1718_swgs_line_ship"
	preferred_attack_range = median				#舰船进攻模式下保持的进攻距离 #PS：舰船处于进攻模式下会保持该距离
	formation_distance = median					#舰船被动模式下保持的攻击距离，如果小于该距离切换为进攻模式 #PS:本切换极度不稳定，建议使用强制切换
	return_to_formation_distance = max			#舰船将强制切换回“passive_move_pattern”的战斗位置距离
	source_target_anchor_factor = 1.0			#目标和锚点之间坐标权重倍率，该值不能为0 不写该条目默认为1，最低值疑似为0.1或者0.01 数值越低锚点离自己舰船越近
	combat_target_anchor = root					#以敌人舰队领袖船为目标锚点 parent是锁定目标敌舰 root是敌人舰队首位船

	collision_awareness = 0.75 					#舰船运动意识，当舰船的移动半径运算结果要撞上去之前，有多高的移动概率，越大的船移动概率越低
	collision_radius = 1.0						#舰船移动半径倍率，与舰船的定义半径进行相乘，这一数值应该优先修正舰船shipsize定义中的半径，使其与舰船的大小相近
	collision_strafe = 2.5						#舰船位置冲突时，舰船会选择多大的角度进行回避，一般情况下拉高点舰船回避的速度越快，容易出现鬼畜，数值要适合
	attack_move_pattern = stay_at_range			#舰船进攻模式
	passive_move_pattern = stay_at_range		#舰船被动模式	
	desc = "FOC_V1718_SWGS_LINE_BEHAVIOR_DESC"
}
