# V235: Resistance is NOT the pre-Treaty Imperial Rebellion.
foc_v235_is_resistance = {
    OR = {
        has_country_flag = foc_khan_first_order_resistance_country
        has_country_flag = foc_republic_origin_post_treaty_rebel_country
        has_country_flag = foc_republic_origin_post_treaty_player_rebels
    }
    NOT = { has_country_flag = foc_republic_origin_imperial_rebel_country }
    NOT = { has_country_flag = foc_republic_origin_rebel_alliance_country }
}

# Ship scope, all hulls owned by the two actual factions, regardless of fleet flags.

    OR = {
        is_ship_size = imperialsd
        is_ship_size = imperialsd_foc_lowest
        is_ship_size = foc_v118_imperialsd_bastion
        is_ship_size = foc_v118_imperialsd_bastion_foc_lowest
    }
}
