can_colonize_planet_trigger = {
	custom_tooltip = {
		fail_text = "COLONIZATION_IMPOSSIBLE_UNDER_INVESTIGATION"
		NOT = { has_carrier_flag = planet_under_investigation }
	}

	custom_tooltip = {
		fail_text = "COLONIZATION_IMPOSSIBLE_DYSON_SPHERE"
		if = {
			limit = {
				is_artificial = no	# Colonizing habitats is fine
			}
			system_has_dyson_sphere = no
		}
	}

	custom_tooltip = {
		fail_text = "COLONIZATION_IMPOSSIBLE_DYSON_SPHERE_CONSTRUCTION"
		if = {
			limit = {
				system_has_dyson_sphere = no
				is_artificial = no	# Colonizing habitats is fine
			}
			NOT = {
				solar_system = {
					has_star_flag = dyson_sphere_construction
				}
			}
		}
	}

	custom_tooltip = {
		fail_text = "COLONIZATION_IMPOSSIBLE_COSMOGENESIS_EXODUS"
		$SCOPE$? = {
			NOT = { owner = { has_event_chain = cosmogenesis_planet_pickup_chain } }
		}
	}

	custom_tooltip = {
		fail_text = "COLONIZATION_IMPOSSIBLE_CRIMSON_CRAWLERS"

		if = {
			limit = {
				has_modifier = crimson_crawlers
			}
			$SCOPE$? = {
				cares_for_crimson_crawlers = yes
			}
		}
	}

	custom_tooltip = {
		fail_text = "COLONIZATION_IMPOSSIBLE_DEFENDED_PLANET"
		NOT = { has_carrier_flag = planet_defended_forbid_colonization }
	}

	custom_tooltip = {
		fail_text = "COLONIZATION_IMPOSSIBLE_SMEEGIBB_PLANET"
		NOT = { has_carrier_flag = cannot_colonize_smeegibb_shelter }
	}

	custom_tooltip = {
		fail_text = "COLONIZATION_IMPOSSIBLE_EMBARKING"
		$SCOPE$? = {
			NOT = {
				owner = { has_country_flag = already_embarking }
			}
		}
	}
}

benefits_from_waystation_network = {
	OR = {
		is_same_value = $NOMAD$ 							# this country IS the Wayline owner.
		has_waystation_pact = $NOMAD$						# Wayline owner and this country have a wayline pact.
		is_in_federation_with = $NOMAD$						# Wayline owner and this country are in a federation together.
		is_overlord_to = $NOMAD$							# This country is the overlord of the wayline owner.
		AND = {
			is_subject = yes
			OR = {
				$NOMAD$ = {
					is_overlord_to = $SYSTEM_OWNER$			# Wayline owner is the overlord of this country.
				}
				AND = {
					exists = $NOMAD$.overlord
					$NOMAD$.overlord = {
						is_overlord_to = $SYSTEM_OWNER$		# This country and the wayline owner share an overlord.
					}
				}
			}
		}
		AND = {												# Either this country or the wayline owner are the Galactic Emperor
			is_galactic_community_member = yes				# and the other is in the Galactic Imperium
			$NOMAD$ = {
				is_galactic_community_member = yes
			}
			OR = {
				is_galactic_emperor = yes
				$NOMAD$ = {
					is_galactic_emperor = yes
				}
			}
		}
	}
}

# Scope: country (the contract operator or a candidate operator).
# $SYSTEM$ = the contract target system: from in possible_operator, fromfrom in abort_trigger.
# Humans always count as reachable - accepting a contract out of reach is the player's own call.
# The AI checks that one of its arkships can plot a hyperlane route to the target system
# (route not discovered, closed borders and a locked system all count as unreachable).
# can_access_system runs full pathfinding, so keep this the last condition in any block.
contract_target_system_reachable = {
	OR = {
		is_ai = no
		any_owned_fleet = {
			is_arkship_fleet = yes
			can_access_system = $SYSTEM$
		}
	}
}

contract_gas_giant_can_be_ignited = {
	# Check the gas giant
	is_planet_class = pc_gas_giant
	is_moon = no
	has_anomaly = no
	NOT = {
		# Check that there's no space-faring colony on a moon
		any_moon = {
			exists = space_owner
			exists = colony
			exists = colony.owner
			colony.owner = {
				is_primitive = no
			}
		}
	}
	# Check the issuer
	if = {
		limit = {
			# This scripted trigger is called from different contexts, so we check where the issuer scope is
			prevprevprev? = {
				is_country_type = default
				is_nomadic = no
			}
			NOT = {
				has_carrier_flag = stellar_ignition_planet@prevprevprev
			}
			OR = {
				# If there's a pre-ftl colony on a moon, check that the issuer doesn't care about pre-FTLs
				prevprevprev = {
					has_policy_flag = interference_aggressive
				}
				NOT = {
					any_moon = {
						exists = colony
						colony.owner = {
							is_primitive = yes
						}
					}
				}
			}
		}
	}
	else = {
		# If the issuer scope is not prevprevprev, it's root
		NOT = {
			has_carrier_flag = stellar_ignition_planet@root
		}
		OR = {
			# If there's a pre-ftl colony on a moon, check that the issuer doesn't care about pre-FTLs
			root = {
				has_policy_flag = interference_aggressive
			}
			NOT = {
				any_moon = {
					exists = colony
					colony.owner = {
						is_primitive = yes
					}
				}
			}
		}
	}
}

# Returns true if the country has access to at least one surveyed, unowned, uninhabitable
# precursor planet. Non-nomads check within-border systems; Nomads check systems adjacent
# to their deployed waystations (own system + neighbors).
# Usage: has_accessible_precursor_system = { PRECURSOR_FLAG = precursor_1 }
has_accessible_precursor_system = {
	OR = {
		AND = {
			is_nomadic = no
			any_system_within_border = {
				has_star_flag = $PRECURSOR_FLAG$
				any_system_planet = {
					uninhabitable_regular_planet = yes
					NOT = { has_carrier_flag = precursor_world }
					has_owner = no
					has_anomaly = no
					is_surveyed = {
						who = root
						status = yes
					}
				}
			}
		}
		AND = {
			is_nomadic = yes
			any_owned_nonprimary_starbase = {
				is_waystation_starbase = yes
				solar_system = {
					OR = {
						AND = {
							has_star_flag = $PRECURSOR_FLAG$
							any_system_planet = {
								uninhabitable_regular_planet = yes
								NOT = { has_carrier_flag = precursor_world }
								has_owner = no
								has_anomaly = no
								is_surveyed = {
									who = root
									status = yes
								}
							}
						}
						any_neighbor_system = {
							has_star_flag = $PRECURSOR_FLAG$
							any_system_planet = {
								uninhabitable_regular_planet = yes
								NOT = { has_carrier_flag = precursor_world }
								has_owner = no
								has_anomaly = no
								is_surveyed = {
									who = root
									status = yes
								}
							}
						}
					}
				}
			}
		}
	}
}
