# Old-save migration from 1.8-2.2. Change only the starbase size/level.
# Existing Arkship modules and buildings remain installed and immediately rebuild
# against the dedicated Supremacy core instead of being overwritten by a forced whole-design replacement.
country_event = {
	id = foc_sai_supremacy_arkship.15
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		any_owned_ship = {
			has_modifier = foc_sai_supremacy_arkship_stats
			OR = {
				is_ship_size = military_arkship_tier_1
				is_ship_size = military_arkship_tier_2
				is_ship_size = military_arkship_tier_3
			}
		}
	}

	immediate = {
		# v17.18i: no global design refresh in the Arkship deploy path.
		every_owned_ship = {
			limit = { has_modifier = foc_sai_supremacy_arkship_stats }
			if = {
				limit = { is_ship_size = military_arkship_tier_1 }
				starbase = { set_starbase_size = foc_sai_supremacy_arkship_tier_1 }
			}
			else_if = {
				limit = { is_ship_size = military_arkship_tier_2 }
				starbase = { set_starbase_size = foc_sai_supremacy_arkship_tier_2 }
			}
			else_if = {
				limit = { is_ship_size = military_arkship_tier_3 }
				starbase = { set_starbase_size = foc_sai_supremacy_arkship_tier_3 }
			}
			set_graphical_culture = military_arkship
			set_ship_flag = foc_sai_supremacy_arkship
			set_ship_flag = foc_sai_supremacy_event_spawned_only
			remove_ship_flag = foc_sai_supremacy_model_tier_1
			remove_ship_flag = foc_sai_supremacy_model_tier_2
			remove_ship_flag = foc_sai_supremacy_model_tier_3
		}
	}
}

# Tier III fixed tracker auto-installer.
# set_starbase_module uses the numeric slot index; the seventh slot is the dedicated
# spinal_mount_1 slot declared by the Tier III Supremacy starbase level.
country_event = {
	id = foc_sai_supremacy_arkship.16
	hide_window = yes
	is_triggered_only = yes

	immediate = {
        # Keep the standalone cleanup of obsolete tracker bonuses; do not stack them with the module.
        if = { limit = { has_modifier = foc_sai_supremacy_hyperspace_tracker_country_active } remove_modifier = foc_sai_supremacy_hyperspace_tracker_country_active }
        remove_country_flag = foc_sai_supremacy_hyperspace_tracker_online
        every_owned_ship = {
            limit = { OR = { is_ship_size = foc_sai_supremacy_arkship_tier_1 is_ship_size = foc_sai_supremacy_arkship_tier_1_foc_lowest is_ship_size = foc_sai_supremacy_arkship_tier_2 is_ship_size = foc_sai_supremacy_arkship_tier_2_foc_lowest is_ship_size = foc_sai_supremacy_arkship_tier_3 is_ship_size = foc_sai_supremacy_arkship_tier_3_foc_lowest } }
            if = { limit = { has_modifier = foc_sai_supremacy_hyperspace_tracker } remove_modifier = foc_sai_supremacy_hyperspace_tracker }
            if = { limit = { has_modifier = foc_sai_supremacy_hyperspace_tracker_active } remove_modifier = foc_sai_supremacy_hyperspace_tracker_active }
            if = {
                limit = { OR = { is_ship_size = foc_sai_supremacy_arkship_tier_3 is_ship_size = foc_sai_supremacy_arkship_tier_3_foc_lowest } exists = starbase }
                starbase = {
                    if = {
                        limit = { owner = { has_technology = tech_arkship_tier_3 } NOT = { has_starbase_module = foc_sai_supremacy_hyperspace_tracker_module } }
                        set_starbase_module = { slot = 7 module = foc_sai_supremacy_hyperspace_tracker_module }
                    }
                    if = { limit = { has_starbase_module = foc_sai_supremacy_hyperspace_tracker_module } owner = { set_country_flag = foc_sai_supremacy_hyperspace_tracker_online } }
                }
            }
        }
}
}

# v17.18i: retained crisis post-init event ID. NEW ships use .21 and the native
# initializer from the standalone reference; old ships do not receive default modules.
ship_event = {
    id = foc_sai_supremacy_arkship.22
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        OR = {
            has_ship_flag = foc_khan_supremacy_flagship
            has_ship_flag = foc_sai_supremacy_event_spawned_only
            has_ship_flag = foc_sai_supremacy_arkship
        }
        OR = {
            OR = { is_ship_size = foc_sai_supremacy_arkship_tier_1 is_ship_size = foc_sai_supremacy_arkship_tier_1_foc_lowest }
            OR = { is_ship_size = foc_sai_supremacy_arkship_tier_2 is_ship_size = foc_sai_supremacy_arkship_tier_2_foc_lowest }
            OR = { is_ship_size = foc_sai_supremacy_arkship_tier_3 is_ship_size = foc_sai_supremacy_arkship_tier_3_foc_lowest }
        }
    }

    immediate = {
        # Source-style invisible module conversion is allowed only immediately
        # after our NEW-ship native initializer. Never repack old player modules.
        if = {
            limit = { has_ship_flag = foc_v1718i_fresh_native_modules exists = starbase }
            owner = { country_event = { id = sai_supremacy_arkship.17 } }
            remove_ship_flag = foc_v1718i_fresh_native_modules
        }
        if = {
            limit = { exists = starbase }
            foc_v1718h_sync_supremacy_build_mode = yes
            if = {
                limit = {
                    has_ship_flag = foc_v1718i_native_init_called
                    starbase = { is_arkship_starbase = yes arkship_starbase_has_shipwright_module = yes }
                }
                set_ship_flag = foc_sai_supremacy_arkship_core_initialized_v65
            }
        }
        else = { log = "FOC_V1718I_STARBASE_MISSING: [This.GetName]; no object recreated." }
        owner = { country_event = { id = foc_v1718i_arkship.10 days = 1 } }
        ship_event = { id = foc_v1718h_arkship.2 days = 1 }
}
}
