# Stage-I detector. The D'Qar Resistance base appears as soon as the First
# Order crisis exists; the Phase-II branch is an old-save fallback.
country_event = {
	id = foc_khan_rebel.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
        # V235b: a successful first foundation is permanent history.
        foc_v235b_dqar_bootstrap_allowed = yes
		NOT = { has_global_flag = foc_khan_crisis_finished }
		NOT = { has_global_flag = foc_khan_rebel_spawned }
		NOT = { has_global_flag = foc_khan_rebel_preparing }
		# V109: after the Galactic Treaty, D'Qar already exists as New Republic
		# territory. It may not become the Resistance until Leia's Bloodline
		# break is announced after the First Order appears.
		OR = {
			NOT = { has_global_flag = foc_republic_origin_galactic_treaty }
			NOT = { any_system = { has_star_flag = foc_khan_rebel_dqar_home } }
			has_global_flag = foc_v109_bloodline_resistance_authorized
		}
		# V229: only a HUMAN player New Republic is the in-place future Resistance.
		# An AI New Republic must not block the standalone D'Qar Resistance used by
		# First Order / Imperial Remnant / other alternate playable routes.
		NOT = {
			AND = {
				has_global_flag = foc_republic_origin_galactic_treaty
				any_playable_country = {
					is_ai = no
					has_country_flag = foc_republic_origin_player_republic
					has_country_flag = foc_sw_new_republic_country
				}
			}
		}
		# Human New Republic continues in place through .197/.198; all other
		# playable routes keep the standalone D'Qar constructor.
		OR = {
			# V185 immediate Leia triggers: FO origin game start or First Order Plan completion.
			has_global_flag = foc_v185_leia_resistance_authorized
			AND = {
				has_global_flag = foc_republic_origin_galactic_treaty
				has_global_flag = foc_v158_leia_resistance_foundation_due
			}
			AND = {
				has_global_flag = foc_khan_phase_one
				exists = event_target:foc_khan_first_order
			}
			AND = {
				has_global_flag = foc_khan_phase_two
				exists = event_target:foc_khan_final_order
			}
			AND = {
				has_country_flag = foc_republic_origin_player_imperial_remnant
				has_country_flag = foc_republic_origin_first_order_plan_completed
				has_country_flag = foc_republic_origin_remnant_rebranded_first_order
				has_global_flag = foc_republic_origin_galactic_treaty
			}
			AND = {
				has_global_flag = fop_origin_player_crisis_exists
				exists = event_target:fop_first_order_crisis_country
				event_target:fop_first_order_crisis_country = {
					has_origin = origin_fop_first_order
					has_country_flag = fop_origin_crisis_active
				}
			}
		}
	}

	immediate = {
        if = {
            limit = { foc_v235b_dqar_bootstrap_allowed = yes }
    		set_global_flag = foc_khan_rebel_preparing
    		log = "FOC_V178_DQAR: Resistance bootstrap started; using the exact V177 map position."

    		# New games have no fixed D'Qar. Old saves adopt the former fixed-map
    		# system, while current games generate it from the Coruscant anchor.
    		foc_v178_adopt_static_dqar = yes
    		if = {
    			limit = { any_system = { has_star_flag = foc_khan_rebel_dqar_home } }
    			random_system = {
    				limit = { has_star_flag = foc_khan_rebel_dqar_home }
    				save_global_event_target_as = foc_khan_rebel_system
    				save_global_event_target_as = foc_khan_rebel_dqar_system
    				random_system_planet = {
    					limit = {
    						OR = {
    							has_planet_flag = foc_khan_rebel_capital
    							has_planet_flag = foc_khan_rebel_dqar
    						}
    					}
    					set_planet_flag = foc_khan_rebel_capital
    					save_global_event_target_as = foc_khan_rebel_capital
    				}
    			}
    		}
    		else = {
    			foc_v178_spawn_dqar_at_map_position = yes
    			if = {
    				limit = { NOT = { exists = event_target:foc_khan_rebel_spawn_anchor } }
    				# The map initializer is asynchronous. Release the lock, but also
    				# explicitly restart Stage I so a one-shot story dispatch cannot be lost.
    				remove_global_flag = foc_khan_rebel_preparing
    				country_event = { id = foc_khan_rebel.1 days = 10 }
    			}
    		}

    		# V203: the bootstrap owns its completion retries. Treaty +23y previously
    		# called only this event, leaving D'Qar permanently stuck in "preparing".
    		# Existing callers may also queue .2; its global guards make those calls
    		# idempotent after the first successful country creation.
    		if = {
    			limit = { has_global_flag = foc_khan_rebel_preparing }
    			country_event = { id = foc_khan_rebel.2 days = 2 }
    			country_event = { id = foc_khan_rebel.2 days = 5 }
    			country_event = { id = foc_khan_rebel.2 days = 15 }
    		}
	
        }
    }
}
