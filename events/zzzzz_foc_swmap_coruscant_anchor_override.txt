# New-game bootstrap. It replaces the on-action entry point without reusing an
# existing event ID, so the layout does not depend on duplicate-event load
# order.
country_event = {
    id = foc_swmap.100
    hide_window = yes
    is_triggered_only = yes
    fire_only_once = yes

    trigger = {
        is_country_type = global_event
        NOT = { has_global_flag = foc_sw_fallen_empires_created }
        NOT = { has_global_flag = foc_sw_fallen_empires_initializing }
    }

    immediate = {
        # The Republic and playable-Hutt origins are the pre-Imperial branches
        # of this map. Mark them before the region exists so monthly migration
        # events (most notably the post-Imperial Executor wreckage site on Jakku)
        # can never run in the short interval before Republic-era setup.
        if = {
            limit = {
                any_playable_country = {
                    is_ai = no
                    OR = {
                        has_origin = origin_foc_republic_unification
                        has_origin = origin_foc_hutt_criminal_empire
                    }
                }
            }
            set_global_flag = foc_republic_origin_bootstrap
            set_global_flag = foc_republic_origin_no_jakku_executor_site
            if = {
                limit = {
                    any_playable_country = {
                        is_ai = no
                        has_origin = origin_foc_hutt_criminal_empire
                    }
                }
                set_global_flag = foc_v197_hutt_republic_era_selected
            }
        }
        set_global_flag = foc_sw_fallen_empires_initializing
        log = "FOC_SWMAP V178: generating the v3 Star Wars region; D'Qar, Crait and Ajan Kloss hidden, Jakku main-galaxy wormhole enabled, outer edge radius 550."
        country_event = { id = foc_swmap.101 }
    }
}
