# 本·索罗在神殿毁灭时仍仅为剧情人物；只有第一秩序真正成立后，此事件才允许在玩家第一秩序上显示。
country_event = {
    id = foc_kylo.10
    title = foc_kylo.10.name
    desc = foc_kylo.10.desc
    picture = GFX_foc_saga_dark_jedi
    show_sound = event_mystic_reveal
    is_triggered_only = yes

    trigger = {
        is_ai = no
        is_regular_empire = yes
        # Ben cannot become Kylo before the canonical public First Order era.
        OR = {
            has_global_flag = foc_khan_phase_one
            has_country_flag = fop_origin_public_emergence
        }
        NOT = { has_global_flag = foc_khan_phase_two }
        NOT = { has_country_flag = foc_saga_kylo_fall_shown }
        OR = {
            has_country_flag = foc_khan_first_order_country
            has_country_flag = fop_player_first_order
            has_country_flag = foc_republic_origin_became_first_order
            has_country_flag = foc_republic_origin_remnant_rebranded_first_order
        }
    }

    immediate = {
        set_country_flag = foc_saga_kylo_fall_shown
        set_global_flag = foc_saga_ben_solo_fallen
        if = {
            limit = {
                OR = {
                    has_country_flag = foc_khan_first_order_country
                    has_country_flag = foc_republic_origin_became_first_order
                    has_country_flag = fop_player_first_order
                    has_country_flag = foc_republic_origin_remnant_rebranded_first_order
                }
            }
            foc_saga_ensure_kylo = yes
        }
        set_timed_country_flag = { flag = foc_saga_kylo_knights_wait days = 720 }
    }

    option = {
        name = foc_kylo.10.a
    }
}

# -----------------------------------------------------------------------------
# 与 foc_story.11（基地毁灭与汉·索罗之死）互补：那一个讲基地，这一个讲那座桥。
country_event = {
    id = foc_kylo.12
    title = foc_kylo.12.name
    desc = foc_kylo.12.desc
    picture = GFX_foc_story_starkiller_destroyed
    show_sound = event_mystic_reveal
    is_triggered_only = yes

    trigger = {
        is_ai = no
        is_regular_empire = yes
        has_country_flag = foc_saga_kylo_fall_shown
        NOT = { has_country_flag = foc_saga_kylo_patricide_shown }
        OR = {
            has_global_flag = foc_khan_starkiller_destroyed
            has_global_flag = foc_story_starkiller_destroyed_shown
        }
    }

    immediate = {
        set_country_flag = foc_saga_kylo_patricide_shown
        set_global_flag = foc_saga_han_solo_dead
        every_country = {
            every_owned_leader = {
                limit = { has_leader_flag = foc_canon_han_solo_leader }
                kill_leader = { show_notification = no }
            }
        }
        clear_global_event_target = foc_canon_han_solo
        if = {
            limit = {
                OR = {
                    has_country_flag = foc_khan_first_order_country
                    has_country_flag = foc_republic_origin_became_first_order
                    has_country_flag = fop_player_first_order
                    has_country_flag = foc_republic_origin_remnant_rebranded_first_order
                }
            }
            foc_saga_ensure_kylo = yes
            foc_saga_kylo_advance = yes
        }
    }

    option = {
        name = foc_kylo.12.a
    }
}
