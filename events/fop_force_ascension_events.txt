# Any ordinary or Fallen Empire whose every colony has been occupied by the
# player crisis is bound as a Force Thrall. This is the conquest mechanism used
# by the ascension objectives.
country_event = {
    id = fop_origin.60
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        exists = event_target:fop_first_order_crisis_country
        event_target:fop_first_order_crisis_country = {
            has_country_flag = fop_origin_crisis_active
        }
        NOT = { is_same_value = event_target:fop_first_order_crisis_country }
        OR = {
            is_regular_empire = yes
            is_country_type = fallen_empire
            is_country_type = awakened_fallen_empire
        }
        is_subject = no
        any_owned_planet = { is_colony = yes }
        NOT = {
            any_owned_planet = {
                is_colony = yes
                NOT = { is_controlled_by = event_target:fop_first_order_crisis_country }
            }
        }
    }

    immediate = {
        if = {
            limit = { is_overlord = yes }
            every_subject = {
                set_subject_of = {
                    who = event_target:fop_first_order_crisis_country
                    preset = preset_foc_khan_satrapy
                }
                set_country_flag = fop_force_bound_subject
                set_timed_country_flag = { flag = rebellion_cooldown days = 3600 }
                add_modifier = { modifier = fop_force_bound_subject_modifier days = -1 }
                set_faction_hostility = {
                    target = event_target:fop_first_order_crisis_country
                    set_hostile = no
                    set_neutral = no
                    set_friendly = yes
                }
                event_target:fop_first_order_crisis_country = {
                    set_faction_hostility = {
                        target = prev
                        set_hostile = no
                        set_neutral = no
                        set_friendly = yes
                    }
                }
            }
        }

        set_subject_of = {
            who = event_target:fop_first_order_crisis_country
            preset = preset_foc_khan_satrapy
        }
        set_country_flag = fop_force_bound_subject
        set_timed_country_flag = { flag = rebellion_cooldown days = 3600 }
        add_modifier = { modifier = fop_force_bound_subject_modifier days = -1 }

        set_faction_hostility = {
            target = event_target:fop_first_order_crisis_country
            set_hostile = no
            set_neutral = no
            set_friendly = yes
        }
        event_target:fop_first_order_crisis_country = {

                target = prev
                set_hostile = no
                set_neutral = no
                set_friendly = yes

        }
    }
}
