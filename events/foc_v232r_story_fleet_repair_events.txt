# Source-tracked old-save migration. Never infer an undocumented choice from its absence.
country_event = {
    id = foc_v232r.3
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        foc_v197_republic_story_actor = yes
        has_country_flag = foc_republic_origin_republic_country
        has_global_flag = foc_republic_origin_separatist_war_active
        NOT = { has_country_flag = foc_republic_origin_separatist_war_won }
        NOT = { has_country_flag = foc_republic_origin_order_66_executed }
        NOT = { has_country_flag = foc_v232r_plot_ledger_initialized }
    }
    immediate = {
        set_country_flag = foc_v232r_plot_ledger_initialized
        set_country_flag = foc_v232r_legacy_plot_migration
        # A completed old bar is not a reason to skip the new battle requirements.
        remove_country_flag = foc_republic_origin_palpatine_plot_complete
        if = {
            limit = { NOT = { any_situation = { is_situation_type = situation_foc_republic_origin_palpatine_plot } } }
            start_situation = { type = situation_foc_republic_origin_palpatine_plot }
        }
        remove_country_flag = foc_v232o_separatist_surrender_offer_pending
        if = { limit = { has_country_flag = foc_republic_origin_naboo_crisis_resolved } set_country_flag = foc_v232r_plot_naboo }
        if = { limit = { has_country_flag = foc_republic_origin_clone_army_plan } set_country_flag = foc_v232r_plot_clone_army }
        if = { limit = { has_country_flag = foc_republic_origin_kuat_investment_done } set_country_flag = foc_v232r_plot_kuat }
        if = { limit = { has_country_flag = foc_saga_cw_coruscant_shown } set_country_flag = foc_v232r_plot_coruscant }
        if = {
            limit = { NOT = { has_country_flag = foc_v232r_plot_choice_12_done } has_country_flag = foc_republic_origin_combines_refused }
            set_country_flag = foc_v232r_plot_choice_12_done
        }
        if = {
            limit = { NOT = { has_country_flag = foc_v232r_plot_choice_13_done } has_country_flag = foc_republic_origin_military_act_passed }
            set_country_flag = foc_v232r_plot_choice_13_done
            set_country_flag = foc_v232r_plot_choice_13_favorable
        }
        if = {
            limit = { NOT = { has_country_flag = foc_v232r_plot_choice_13_done } has_country_flag = foc_republic_origin_military_act_blocked }
            set_country_flag = foc_v232r_plot_choice_13_done
        }
        if = {
            limit = { NOT = { has_country_flag = foc_v232r_plot_choice_14_done } has_country_flag = foc_republic_origin_audit_pursued }
            set_country_flag = foc_v232r_plot_choice_14_done
        }
        if = {
            limit = { NOT = { has_country_flag = foc_v232r_plot_choice_15_done } has_country_flag = foc_republic_origin_emergency_powers_extended }
            set_country_flag = foc_v232r_plot_choice_15_done
            set_country_flag = foc_v232r_plot_choice_15_favorable
        }
        if = {
            limit = { NOT = { has_country_flag = foc_v232r_plot_choice_15_done } has_country_flag = foc_republic_origin_emergency_powers_lapsed }
            set_country_flag = foc_v232r_plot_choice_15_done
        }
        if = {
            limit = { NOT = { has_country_flag = foc_v232r_plot_choice_8_done } has_country_flag = foc_republic_origin_jedi_investigation_refused }
            set_country_flag = foc_v232r_plot_choice_8_done
            set_country_flag = foc_v232r_plot_choice_8_favorable
        }
        if = {
            limit = { NOT = { has_country_flag = foc_v232r_plot_choice_8_done } has_country_flag = foc_republic_origin_jedi_investigation_backed }
            set_country_flag = foc_v232r_plot_choice_8_done
        }
        if = { limit = { NOT = { has_country_flag = foc_v232r_plot_choice_12_done } } country_event = { id = foc_v232r.112 days = 1 } }
        if = { limit = { NOT = { has_country_flag = foc_v232r_plot_choice_13_done } } country_event = { id = foc_v232r.113 days = 2 } }
        if = { limit = { NOT = { has_country_flag = foc_v232r_plot_choice_14_done } } country_event = { id = foc_v232r.114 days = 3 } }
        if = { limit = { NOT = { has_country_flag = foc_v232r_plot_choice_15_done } } country_event = { id = foc_v232r.115 days = 4 } }
        if = { limit = { NOT = { has_country_flag = foc_v232r_plot_choice_8_done } } country_event = { id = foc_v232r.108 days = 5 } }
        foc_v232r_recalculate_plot = yes
        log = "FOC_V232R_PLOT: legacy wartime score rebuilt from recorded decisions; undocumented choices require a score-only review." 
    }
}

# One global monthly/load entry; no recurring chain per country outside the actual raid.
event = {
    id = foc_v232r.10
    hide_window = yes
    is_triggered_only = yes
    immediate = {
        every_country = {
            if = {
                limit = {
                    foc_v197_republic_story_actor = yes
                    has_country_flag = foc_republic_origin_republic_country
                    has_global_flag = foc_republic_origin_separatist_war_active
                    NOT = { has_country_flag = foc_republic_origin_separatist_war_won }
                    NOT = { has_country_flag = foc_republic_origin_order_66_executed }
                }
                country_event = { id = foc_v232r.3 }
                if = {
                    limit = { has_country_flag = foc_v232r_plot_ledger_initialized }
                    foc_v232r_recalculate_plot = yes
                }
                if = {
                    limit = {
                        has_country_flag = foc_saga_cw_coruscant_assault_launched
                        NOT = { has_country_flag = foc_saga_cw_coruscant_shown }
                        NOT = { has_country_flag = foc_v232r_coruscant_poll_running }
                    }
                    set_country_flag = foc_v232r_coruscant_poll_running
                    country_event = { id = foc_v232r.1 days = 1 }
                }
                # Old .3 completion without true ruler death: kill the marked actor once.
                if = {
                    limit = {
                        has_country_flag = foc_saga_cw_coruscant_shown
                        NOT = { has_country_flag = foc_v232r_dooku_cleanup_checked }
                    }
                    set_country_flag = foc_v232r_dooku_cleanup_checked
                    foc_v232r_kill_dooku_at_coruscant = yes
                    every_country = {
                        limit = { has_country_flag = foc_republic_origin_separatist_country }
                        country_event = { id = foc_v232r.2 days = 1 }
                    }
                }
            }
        }
    }
}
