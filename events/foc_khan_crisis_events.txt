# Galactic Community detector.
# FOC 2.72: the Community awakens the New Republic and Imperial Remnant.
# First Order is spawned when either side is eliminated or their War in Heaven
# ends in peace. It then absorbs any surviving Imperial Remnant.
country_event = {
    id = foc_khan.1
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        is_ai = no
        has_global_flag = foc_sw_fallen_empires_ready
        NOT = { has_global_flag = foc_sw_war_in_heaven_started }
        NOT = { has_global_flag = foc_sw_war_in_heaven_scheduled }
        exists = event_target:foc_sw_new_republic
        exists = event_target:foc_sw_imperial_remnant
        OR = {
            # Republic saga: exactly twenty-four years after the Galactic Treaty.
            AND = {
                OR = {
                    has_global_flag = foc_republic_origin_galactic_treaty
                    has_global_flag = fop_origin_treaty_epoch
                }
                has_global_flag = foc_v112_treaty_wih_clock_started
                NOT = { has_global_flag = foc_v112_treaty_wih_24y_delay }
            }
            # First Order origin and all ordinary origins: sixty years after the
            # selected chronology start (2200 default; controller can delay it).
            AND = {
                NOT = { has_global_flag = foc_republic_origin_galactic_treaty }
                foc_story_war_in_heaven_due = yes
            }
        }
    }

    immediate = {
        set_global_flag = foc_sw_war_in_heaven_scheduled
        event_target:foc_sw_new_republic = {
            country_event = { id = foc_fallen_war.10 days = 1 }
        }
    }
}

# Spawn Ilum from a real system scope, following the Khan crisis pattern.
country_event = {
    id = foc_khan.2
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        NOT = { has_global_flag = foc_khan_crisis_started }
        NOT = { has_global_flag = foc_khan_crisis_disabled }
        # V159 hard guard: player-owned First Order paths never spawn a second
        # NPC First Order, even if another script clears the global disable flag.
        NOT = {
            any_playable_country = {
                is_ai = no
                OR = {
                    has_origin = origin_fop_first_order
                    has_country_flag = foc_republic_origin_player_empire
                    has_country_flag = foc_republic_origin_player_imperial_remnant
                    has_country_flag = foc_republic_origin_empire_crisis
                    has_country_flag = fop_player_first_order
                    has_country_flag = foc_republic_origin_became_first_order
                    has_country_flag = foc_republic_origin_remnant_rebranded_first_order
                }
            }
        }
    }

    immediate = {
        remove_global_flag = foc_khan_crisis_spawn_scheduled
        set_global_flag = foc_khan_crisis_started
        set_global_flag = foc_khan_phase_one_preparing

        # V112: crisis Ilum uses the same galaxy-core-relative location as the
        # playable First Order origin instead of sitting next to Jakku.
        set_spawn_system_batch = begin
        no_scope = {
            spawn_system = {
                hyperlane = no
                is_discovered = no
                min_distance >= 550
                max_distance <= 560
                min_orientation_angle = 142
                max_orientation_angle = 148
                initializer = foc_khan_ilum_system
            }
        }
        set_spawn_system_batch = end

        country_event = { id = foc_khan.3 days = 3 }
    }
}

# Find the newly spawned system by its initializer flag.
country_event = {
	id = foc_khan.3
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_global_flag = foc_khan_phase_one_preparing
	}

	immediate = {
		random_system = {
			limit = {
				has_star_flag = foc_khan_ilum_home
			}
			save_global_event_target_as = foc_khan_first_order_system

			random_system_planet = {
				limit = {
					has_planet_flag = foc_khan_starkiller_anchor
				}
				save_global_event_target_as = foc_khan_starkiller_anchor
			}
		}

		# Remove ordinary lanes before the First Order receives fleets. Its
		# exclusive gateway pair is created as soon as the crisis country exists.
		country_event = { id = foc_khan.1710 }

		if = {
			limit = {
				exists = event_target:foc_khan_first_order_system
				exists = event_target:foc_khan_starkiller_anchor
			}
			country_event = {
				id = foc_khan.4
				days = 1
			}
		}
		else = {
			# Retry without duplicating the system.
			country_event = {
				id = foc_khan.3
				days = 5
			}
		}
	}
}

# Keep crisis Exegol inside a vanilla Electric Storm. Because Exegol has no
# ordinary hyperlanes, the storm remains confined to the hidden system; if the
# engine ends it, this restores it on the next monthly pass.
# 3.11w-fix: this was a country_event gated on is_country_type = global_event
# and registered on on_monthly_pulse_country — but that pulse never runs for
# the global_event country type (it has no pulse events; the same engine rule
# this MOD documents for its own crisis types). The storm was therefore never
# created. It is now a scopeless event on the plain monthly pulse.
event = {
	id = foc_khan.1711
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		host_has_dlc = "Cosmic Storms"
		OR = {
			has_global_flag = foc_khan_phase_two_preparing
			has_global_flag = foc_khan_phase_two
		}
		NOT = { has_global_flag = foc_khan_crisis_finished }
		any_system = { has_star_flag = foc_khan_exegol_home }
	}

	immediate = {
		if = {
			limit = { NOT = { exists = event_target:foc_khan_final_order_system } }
			random_system = {
				limit = { has_star_flag = foc_khan_exegol_home }
				save_global_event_target_as = foc_khan_final_order_system
			}
		}

		# 3.11w-2: the diagnostics showed the engine accepts the storm but it
		# does not end up covering Exegol (start position is snapped elsewhere
		# and the storm drifts). A permanent storm influence field centered on
		# Exegol is the engine's own mechanism for attracting and holding
		# storms over a system; create it once.
		if = {
			limit = {
				NOT = { has_global_flag = foc_khan_exegol_storm_field_created_311w }
				exists = event_target:foc_khan_final_order_system
			}
			create_cosmic_storm_influence_field = {
				center = event_target:foc_khan_final_order_system
				radius = { base = 60 }
			}
			set_global_flag = foc_khan_exegol_storm_field_created_311w
			log = "FOC_3.11W: created a permanent storm influence field on crisis Exegol."
		}

		if = {
			limit = {
				exists = event_target:foc_khan_final_order_system
				event_target:foc_khan_final_order_system = { is_inside_storm = no }
			}
			# 3.11w-3: sixteen consecutive months of diagnostics proved the
			# engine accepts create_cosmic_storm but snaps the storm to its own
			# placement grid, which never includes the hyperlane-isolated
			# Exegol; every attempt just released a stray storm into the
			# galaxy. The monthly creation loop is therefore removed. Exegol
			# instead carries a permanent system modifier that replicates the
			# vanilla Electric Storm's ship effects (+100% sublight speed,
			# halved shields). The influence field above still attracts any
			# naturally wandering storm toward Exegol as flavor.
			event_target:foc_khan_final_order_system = {
				if = {
					limit = {
						NOT = { has_modifier = foc_khan_exegol_electric_storm_311w }
					}
					add_modifier = {
						modifier = foc_khan_exegol_electric_storm_311w
						days = -1
					}
					log = "FOC_3.11W: applied the permanent Electric Storm system modifier to crisis Exegol."
				}
			}
		}
	}
}

# 1.02 migration: remove target markers left permanently by the brief 1.01 build.
# Wait until no current Starkiller firing sequence is active so an in-progress shot
# loaded from a 1.01 save keeps its target animation until normal completion.
event = {
	id = foc_khan.38
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		NOT = { has_global_flag = foc_khan_102_scar_migration_done }
		NOT = { exists = event_target:myk_starkiller_target_system }
		NOT = { has_global_flag = foc_khan_forced_starkiller_in_progress }
	}

	immediate = {
		every_country = {
			every_owned_megastructure = {
				limit = { has_megastructure_flag = myk_starkiller_persistent_target_scar }
				remove_megastructure = this
			}
		}
		set_global_flag = foc_khan_102_scar_migration_done
	}
}

# 3.11w-v3: link the Supremacy escort to the flagship once both fleets exist.
# Doing this two days after creation avoids referencing a half-built fleet,
# which is why the earlier in-line auto_follow_fleet never took effect.
country_event = {
	id = foc_khan.4701
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				any_owned_fleet = { has_fleet_flag = foc_khan_supremacy_fleet }
				any_owned_fleet = { has_fleet_flag = foc_khan_supremacy_escort_fleet }
			}
			random_owned_fleet = {
				limit = { has_fleet_flag = foc_khan_supremacy_fleet }
				save_event_target_as = foc_khan_supremacy_fleet_link
			}
			every_owned_fleet = {
				limit = { has_fleet_flag = foc_khan_supremacy_escort_fleet }
				auto_follow_fleet = { target = event_target:foc_khan_supremacy_fleet_link attack_fleet = no }
				log = "FOC_3.11W: Supremacy escort fleet linked and now following the flagship."
			}
		}
		else = {
			log = "FOC_3.11W: Supremacy escort link FAILED - flagship or escort fleet is missing."
		}
	}
}
