namespace = foc_khan_204

# Existing pre-2.04 saves may contain the retired one-ship Xyston fleets.
# FOC 3.11t hands all such saves to the consolidated Exegol-guard migration.
event = {
	id = foc_khan_204.1
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		has_global_flag = foc_khan_phase_two
		NOT = { has_global_flag = foc_khan_crisis_finished }
		exists = event_target:foc_khan_final_order
		exists = event_target:foc_khan_final_order_system
		exists = event_target:foc_khan_exegol_capital
		event_target:foc_khan_final_order = {
			has_country_flag = foc_khan_final_order_country
			NOT = { has_country_flag = foc_khan_final_initial_fleet_204_done }
		}
	}
	immediate = {
		event_target:foc_khan_final_order = {
			set_country_flag = foc_khan_final_initial_fleet_204_done
			country_event = { id = foc_final_guard.1 days = 1 }
		}
	}
}
