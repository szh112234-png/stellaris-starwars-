namespace = foc_v1718k_supply

# Country game-start and explicit load synchronization. No up-front payment.
country_event = {
    id = foc_v1718k_supply.1
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        OR = {
            has_origin = origin_fop_first_order
            has_modifier = foc_v1718k_secret_resource_collection
        }
    }
    immediate = { foc_v1718k_sync_secret_resource_collection = yes }
}

# Exactly one monthly payout path. Reuse the mod's fixed event-income convention.
country_event = {
    id = foc_v1718k_supply.2
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        OR = {
            AND = {
                has_origin = origin_fop_first_order
                NOT = { has_country_flag = foc_v1718k_secret_resource_collection_ended }
            }
            has_modifier = foc_v1718k_secret_resource_collection
        }
    }
    immediate = {
        foc_v1718k_sync_secret_resource_collection = yes
        if = {
            limit = {
                has_origin = origin_fop_first_order
                has_modifier = foc_v1718k_secret_resource_collection
                NOT = { has_country_flag = foc_v1718k_secret_resource_collection_ended }
                NOT = { has_country_flag = foc_v1718k_secret_resource_month_paid }
            }
            # The ordinary monthly hook fires every game month; a saved 29-day
            # guard prevents duplicate same-tick callbacks without blocking next month.
            set_timed_country_flag = { flag = foc_v1718k_secret_resource_month_paid days = 29 }
            add_resource = {
                sr_tibanna_swgs = 100
                sr_coaxium_foc = 100
            }
        }
    }
}

# Standard completion hooks: root = builder/owner country, from = megastructure.
# Construction sites and upgrades still in progress do NOT end the allowance.
country_event = {
    id = foc_v1718k_supply.3
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        has_origin = origin_fop_first_order
        exists = from
        from = {
            OR = {
                is_megastructure_type = fop_starkiller_base_final
                is_megastructure_type = myk_starkiller_base_final
            }
            is_upgrading = no
        }
    }
    immediate = {
        set_country_flag = foc_v1718k_secret_resource_collection_ended
        if = {
            limit = { has_modifier = foc_v1718k_secret_resource_collection }
            remove_modifier = foc_v1718k_secret_resource_collection
        }
    }
}

# Single-player load only: no income, no debt/back-pay, no reinitialization of ships.
event = {
    id = foc_v1718k_supply.90
    hide_window = yes
    is_triggered_only = yes
    immediate = {
        every_country = {
            limit = {
                OR = {
                    has_origin = origin_fop_first_order
                    has_modifier = foc_v1718k_secret_resource_collection
                }
            }
            country_event = { id = foc_v1718k_supply.1 }
        }
    }
}
