# V232q: actual-operation based repair, including old finished objects that
# remained on the network and stale global target references.
country_event = {
    id = foc_gcw_active.91
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        is_ai = no
        OR = {
            has_country_flag = foc_gcw_active_uprising_in_progress
            has_country_flag = foc_gcw_active_uprising_starting
            any_espionage_operation = { has_espionage_type = operation_foc_gcw_instigate_system_uprising }
        }
    }
    immediate = {
        # Safe old-save migration: require the resolved flag or the old finale
        # cleanup signature. Do not infer completion from a chapter index alone.
        every_espionage_operation = {
            limit = { has_espionage_type = operation_foc_gcw_instigate_system_uprising }
            if = {
                limit = {
                    OR = {
                        has_espionage_operation_flag = foc_v232q_coup_resolved
                        AND = {
                            owner = {
                                NOT = { has_country_flag = foc_gcw_active_uprising_in_progress }
                                NOT = { has_country_flag = foc_gcw_active_uprising_starting }
                            }
                            NOT = { exists = event_target:foc_gcw_active_uprising_target }
                            NOT = { exists = event_target:foc_gcw_active_uprising_target_system }
                        }
                    }
                }
                foc_gcw_active_uprising_cleanup = yes
                log = "FOC_V232Q_COUP: retired a completed legacy operation; no new uprising or reward was granted."
                destroy_espionage_operation = this
            }
            else = {
                # Recover only the pointer to a still-live legacy operation.
                save_global_event_target_as = foc_v232q_active_coup_operation
                owner = { set_country_flag = foc_gcw_active_uprising_in_progress }
            }
        }
        foc_v232q_cleanup_orphaned_coup = yes
    }
}
