# Route/relic initializer. One human dispatcher only; all actions are idempotent.
country_event = {
    id = foc_phase1.1
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        is_ai = no
        NOT = { has_global_flag = foc_phase1_trade_corridors_initialized }
        any_system = { has_star_flag = foc_swmap_corellia_home }
    }
    immediate = {
        # Corellian commercial spine: Coruscant -> Corellia -> Tatooine/Naboo market arc.
        every_system = {
            limit = { OR = { has_star_flag = foc_sw_coruscant_home has_star_flag = foc_swmap_corellia_home has_star_flag = foc_swmap_tatooine_home has_star_flag = foc_swmap_naboo_home } }

                limit = { is_colony = yes }
                if = { limit = { NOT = { has_modifier = foc_phase1_corellian_commercial_spine } } add_modifier = { modifier = foc_phase1_corellian_commercial_spine days = -1 } }
            }
        }
        # Coreward industrial corridor: Coruscant, Kuat and Fondor are the shipbuilding/administrative axis.
        every_system = {
            limit = { OR = { has_star_flag = foc_sw_coruscant_home has_star_flag = foc_sw_kuat_home has_star_flag = foc_swfill_fondor_home } }
            every_system_planet = {
                limit = { is_colony = yes }
                if = { limit = { NOT = { has_modifier = foc_phase1_coreward_industrial_corridor } } add_modifier = { modifier = foc_phase1_coreward_industrial_corridor days = -1 } }
            }
        }
        # Outer Rim military supply lane: key war worlds, not a copied named-route list.
        every_system = {
            limit = { OR = { has_star_flag = foc_swmap_kashyyyk_home has_star_flag = foc_swmap_mon_cala_home has_star_flag = foc_swmap_scarif_home has_star_flag = foc_sw_jakku_home } }
            every_system_planet = {
                limit = { is_colony = yes }
                if = { limit = { NOT = { has_modifier = foc_phase1_outer_rim_supply_lane } } add_modifier = { modifier = foc_phase1_outer_rim_supply_lane days = -1 } }
            }
        }
        set_global_flag = foc_phase1_trade_corridors_initialized
        set_country_flag = foc_phase1_trade_corridors_seen
        country_event = { id = foc_phase1.60 days = 1 }
    }
}

# Strategic route accounting. V184 uses the same 39-system major-route set as
# the prebuilt vanilla Hyper Relays. Control tiers preserve the old 25/50/75%
# meaning: 10/20/30 of 39 mapped major-route systems. This is dispatched by the
# yearly country pulse, not by a monthly repair/backfill loop.
country_event = {
    id = foc_phase1.61
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        OR = {
            has_country_flag = foc_republic_origin_empire_country
            has_country_flag = foc_republic_origin_imperial_rebel_country
            has_country_flag = foc_republic_origin_rebel_alliance_country
            has_country_flag = foc_khan_rebel_alliance_country
            has_country_flag = foc_republic_origin_post_treaty_rebel_country
        }
    }
    immediate = {
        # Migrate the old single-line archive modifier from earlier Phase I builds.
        remove_modifier = foc_phase1_fleet_archive_savings
        if = {
            limit = { has_country_flag = foc_phase1_venator_retired NOT = { has_modifier = foc_phase1_archive_republic_line } }
            add_modifier = { modifier = foc_phase1_archive_republic_line days = -1 }
        }

        # Count actual ownership of the current major-route network.
        set_variable = { which = foc_phase1_route_score value = 0 }
        every_system = {
            limit = {
                foc_v184_is_major_route_system = yes
                exists = space_owner
                space_owner? = { is_same_value = root }
            }
            root = { change_variable = { which = foc_phase1_route_score value = 1 } }
        }

        if = {
            limit = {
                OR = {
                    has_country_flag = foc_republic_origin_imperial_rebel_country
                    has_country_flag = foc_republic_origin_rebel_alliance_country
                    has_country_flag = foc_khan_rebel_alliance_country
                    has_country_flag = foc_republic_origin_post_treaty_rebel_country
                }
            }

            remove_modifier = foc_phase1_rebel_route_network_1
            remove_modifier = foc_phase1_rebel_route_network_2
            remove_modifier = foc_phase1_rebel_route_network_3

            if = {
                limit = { check_variable = { which = foc_phase1_route_score value >= 30 } }
                add_modifier = { modifier = foc_phase1_rebel_route_network_3 days = -1 }
                if = {
                    limit = { has_country_flag = foc_gcw_ledger_initialized }
                    change_variable = { which = foc_gcw_supply value = 4 }
                    change_variable = { which = foc_gcw_intelligence value = 3 }
                    change_variable = { which = foc_gcw_readiness value = 2 }
                    foc_gcw_clamp_rebel_ledger = yes
                }
            }
            else_if = {
                limit = { check_variable = { which = foc_phase1_route_score value >= 20 } }
                add_modifier = { modifier = foc_phase1_rebel_route_network_2 days = -1 }
                if = {
                    limit = { has_country_flag = foc_gcw_ledger_initialized }
                    change_variable = { which = foc_gcw_supply value = 2 }
                    change_variable = { which = foc_gcw_intelligence value = 2 }
                    foc_gcw_clamp_rebel_ledger = yes
                }
            }
            else_if = {
                limit = { check_variable = { which = foc_phase1_route_score value >= 10 } }
                add_modifier = { modifier = foc_phase1_rebel_route_network_1 days = -1 }
                if = {
                    limit = { has_country_flag = foc_gcw_ledger_initialized }
                    change_variable = { which = foc_gcw_supply value = 1 }
                    foc_gcw_clamp_rebel_ledger = yes
                }
            }
        }
        else_if = {
            limit = { has_country_flag = foc_republic_origin_empire_country }

            remove_modifier = foc_phase1_imperial_route_network_1
            remove_modifier = foc_phase1_imperial_route_network_2
            remove_modifier = foc_phase1_imperial_route_network_3

            if = {
                limit = { check_variable = { which = foc_phase1_route_score value >= 30 } }
                add_modifier = { modifier = foc_phase1_imperial_route_network_3 days = -1 }
                if = {
                    limit = { has_country_flag = foc_gcw_empire_ledger_initialized }
                    change_variable = { which = foc_gcw_control value = 3 }
                    change_variable = { which = foc_gcw_morale value = 2 }
                    foc_gcw_clamp_empire_ledger = yes
                    foc_gcw_refresh_empire_state = yes
                }
            }
            else_if = {
                limit = { check_variable = { which = foc_phase1_route_score value >= 20 } }
                add_modifier = { modifier = foc_phase1_imperial_route_network_2 days = -1 }
                if = {
                    limit = { has_country_flag = foc_gcw_empire_ledger_initialized }
                    change_variable = { which = foc_gcw_control value = 2 }
                    foc_gcw_clamp_empire_ledger = yes
                    foc_gcw_refresh_empire_state = yes
                }
            }
            else_if = {
                limit = { check_variable = { which = foc_phase1_route_score value >= 10 } }
                add_modifier = { modifier = foc_phase1_imperial_route_network_1 days = -1 }
                if = {
                    limit = { has_country_flag = foc_gcw_empire_ledger_initialized }
                    change_variable = { which = foc_gcw_control value = 1 }
                    foc_gcw_clamp_empire_ledger = yes
                    foc_gcw_refresh_empire_state = yes
                }
            }
        }
    }
}
