# The situation itself has zero monthly progress. Every visible story event
# advances it. 8 + 7 + 10 + 10 + 15 + 15 + 8 + 7 + 8 + 7 + 3 + 2 = 100.

country_event = {
    id = foc_sepplot.1
    title = foc_sepplot.1.name
    desc = foc_sepplot.1.desc
    picture = GFX_foc_sep_dark_jedi
    is_triggered_only = yes
    force_open = yes

    trigger = {
        foc_v197_republic_story_actor = yes
        has_country_flag = foc_republic_origin_republic_country
        has_country_flag = foc_republic_origin_naboo_crisis_resolved
        has_country_flag = foc_sepplot_naboo_registered
        NOT = { has_country_flag = foc_sepplot_sith_apprentice_seen }
        NOT = { has_country_flag = foc_republic_origin_separatist_war_started }
    }

    immediate = {
        set_country_flag = foc_sepplot_sith_apprentice_seen
        random_situation = {
            limit = { is_situation_type = situation_foc_republic_origin_separatist_crisis }
            add_situation_progress = 7
        }
        country_event = { id = foc_sepplot.2 days = 720 }
    }
    option = { name = foc_sepplot.1.a }
}

# Old-save / interrupted-chain keeper. It only creates the new situation; once
# Naboo has already happened it registers that first eight points and resumes at
# the Sith-apprentice event. Existing Clone Wars saves are never rewound.
country_event = {
    id = foc_sepplot.90
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        foc_v197_republic_story_actor = yes
        has_country_flag = foc_republic_origin_republic_country
        has_country_flag = foc_republic_origin_initialized
        NOT = { has_country_flag = foc_republic_origin_separatist_war_started }
        NOT = { has_country_flag = foc_republic_origin_arc_completed }
    }

    immediate = {
        if = {
            limit = { NOT = { has_country_flag = foc_republic_origin_separatist_crisis_started } }
            start_situation = { type = situation_foc_republic_origin_separatist_crisis }
        }
        if = {
            limit = {
                has_country_flag = foc_republic_origin_naboo_crisis_resolved
                NOT = { has_country_flag = foc_sepplot_naboo_registered }
            }
            set_country_flag = foc_sepplot_naboo_registered
            random_situation = {
                limit = { is_situation_type = situation_foc_republic_origin_separatist_crisis }
                add_situation_progress = 8
            }
            country_event = { id = foc_sepplot.1 days = 1 }
        }
    }
}
