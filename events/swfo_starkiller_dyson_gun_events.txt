        if = {
            limit = { exists = from }
            from = {
                save_global_event_target_as = myk_starkiller_base_megastructure
                remove_megastructure_flag = myk_starkiller_player_shot_cancelled
                # 1.89: myk_starkiller_player_busy is NO LONGER set here.
                # Setting it during on_started made the action's own `possible`
                # check fail while the one-day selector shell was still running,
                # so the shell never completed and the native 180-day cooldown
                # bar never appeared. The engine already blocks re-use while the
                # action is in progress and while its cooldown is ticking; the
                # busy flag is now set only at impact (event 8000) to cover the
                # post-impact recharge window.
                owner = { save_global_event_target_as = myk_starkiller_base_owner }
                solar_system = {
                    save_global_event_target_as = myk_starkiller_base_system
                    random_system_planet = {
                        limit = {
                            is_star = no
                            OR = {
                                has_planet_flag = myk_starkiller_base_planet_flag
                                has_planet_flag = foc_khan_starkiller_anchor
                            }
                        }
                        save_global_event_target_as = myk_starkiller_base_planet
                    }
                }
            }
        }

        if = {
            limit = { exists = event_target:myk_starkiller_base_owner }

# Monthly old-save repair for the 1.76/1.77 crisis-capital state collision.
# If the shared target scope was erased by the old crisis cleanup while a player
# base kept its busy/scheduled flags, remove only those stale flags. A legitimate
# post-impact cooldown is protected by myk_starkiller_player_cooldown_active.
event = {
    id = myk_starkiller_dyson.8007
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        if = {
            limit = { NOT = { exists = event_target:myk_starkiller_target_system } }
            every_country = {
                every_owned_megastructure = {
                    limit = {
                        OR = {
                            has_megastructure_flag = myk_starkiller_player_busy
                            has_megastructure_flag = myk_starkiller_player_shot_scheduled
                        }
                        NOT = { has_megastructure_flag = myk_starkiller_player_cooldown_active }
                    }
                    remove_megastructure_flag = myk_starkiller_player_busy
                    remove_megastructure_flag = myk_starkiller_player_shot_cancelled
                    remove_megastructure_flag = myk_starkiller_player_shot_scheduled
                }
            }
        }
    }
}
