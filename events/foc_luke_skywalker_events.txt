# Treaty +23 years: Ben destroys the Ossus temple. He remains a narrative-only
# character and does NOT become a recruitable Kylo Ren until the First Order
# actually appears. Leia's political break and Resistance foundation happen in
# the same Treaty year.
country_event = {
    id = foc_luke.21
    title = foc_luke.21.name
    desc = foc_luke.21.desc
    picture = GFX_foc_saga_temple_burns
    location = event_target:foc_swmap_ossus_system
    show_sound = event_mystic_reveal
    is_triggered_only = yes

    trigger = {
        is_regular_empire = yes
        OR = {
            has_global_flag = foc_republic_origin_galactic_treaty
            has_global_flag = fop_origin_treaty_epoch
        }
        has_global_flag = foc_v158_treaty_sequel_timeline_started
        NOT = { has_global_flag = foc_v158_treaty_temple_23y_delay }
        NOT = { has_country_flag = foc_saga_luke_temple_burned_shown }
        NOT = { has_global_flag = foc_saga_luke_died }
        has_country_flag = foc_saga_luke_new_order_shown
        OR = {
            has_country_flag = foc_republic_origin_republic_country
            has_country_flag = foc_sw_new_republic_country
            has_country_flag = foc_republic_origin_post_treaty_rebel_country
            has_country_flag = foc_khan_first_order_resistance_country
        }
    }

    immediate = {
        set_country_flag = foc_saga_luke_temple_burned_shown
        set_global_flag = foc_saga_jedi_temple_burned
        set_global_flag = foc_saga_ben_solo_fallen
        set_global_flag = foc_v158_leia_resistance_foundation_due
        set_global_flag = foc_v109_bloodline_resistance_authorized

        # Dedicated FO origin: Bloodline belongs to the Treaty +23y political
        # break, not to game start. This may create/repair Leia's Resistance,
        # while the separate crisis-arrival event remains disabled for the player.
        every_playable_country = {
            limit = {
                is_ai = no
                has_origin = origin_fop_first_order
                NOT = { has_country_flag = fop_origin_bloodline_shown }
            }
            country_event = { id = fop_origin.33 days = 1 }
        }
        if = {
            limit = { exists = event_target:foc_swmap_ossus_system }
            event_target:foc_swmap_ossus_system = {
                every_system_planet = {
                    limit = { has_planet_flag = foc_swmap_ossus }
                    remove_modifier = foc_canon_ossus_jedi_temple
                    add_modifier = { modifier = foc_canon_ossus_temple_ruins days = -1 }
                    set_planet_flag = foc_canon_jedi_temple_ruins
                }
            }
        }
        remove_modifier = foc_saga_new_jedi_order

        # Human New Republic inherits into Leia's Resistance in place. AI New
        # Republic uses the ordinary D'Qar country constructor. The old
        # First-Order-arrival constructor remains as a fallback if this fails.
        if = {
            limit = {
                is_ai = no
                has_country_flag = foc_republic_origin_player_republic
                has_country_flag = foc_sw_new_republic_country
                NOT = { has_country_flag = foc_republic_origin_dqar_inheritance_complete }
            }
            country_event = { id = foc_republic_origin.197 days = 1 }
        }
        else_if = {
            limit = {
                is_ai = yes
                has_country_flag = foc_sw_new_republic_country
                NOT = { has_global_flag = foc_khan_rebel_spawned }
                NOT = { has_global_flag = foc_khan_rebel_preparing }
            }
            country_event = { id = foc_khan_rebel.1 days = 1 }
        }

        set_timed_country_flag = { flag = foc_saga_luke_ahchto_wait days = 30 }
    }

    option = {
        name = foc_luke.21.a
        add_modifier = { modifier = foc_saga_broken_master days = -1 }
    }
}

            country_event = { id = foc_luke.15 days = 1 }

    }
