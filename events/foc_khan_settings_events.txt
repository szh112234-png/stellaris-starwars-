# Three-tier / independent-fleet-count migration.
# Old five-tier mapping:
# Lowest -> Vanilla combat + Small fleet set
# Initial -> Intermediate combat + Standard fleet set
# Intermediate -> Intermediate combat + Large fleet set
# High -> Intermediate combat + Maximum fleet set
# Madness -> ACOT combat + Maximum fleet set
country_event = {
    id = foc_khan_settings.4
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        NOT = { has_global_flag = foc_khan_three_tier_split_migrated_321 }
        OR = {
            has_global_flag = foc_khan_strength_lowest
            has_global_flag = foc_khan_strength_initial
            has_global_flag = foc_khan_strength_intermediate
            has_global_flag = foc_khan_strength_high
            has_global_flag = foc_khan_strength_madness
            has_global_flag = foc_khan_crisis_started
        }
    }

    immediate = {
        # Preserve the old quantity represented by the former combat tier.
        if = {
            limit = {
                NOR = {
                    has_global_flag = foc_khan_fleet_scale_low
                    has_global_flag = foc_khan_fleet_scale_standard
                    has_global_flag = foc_khan_fleet_scale_large
                    has_global_flag = foc_khan_fleet_scale_max
                }
            }
            if = {
                limit = { has_global_flag = foc_khan_strength_lowest }
                set_global_flag = foc_khan_fleet_scale_low
            }
            else_if = {
                limit = { has_global_flag = foc_khan_strength_initial }
                set_global_flag = foc_khan_fleet_scale_standard
            }
            else_if = {
                limit = { has_global_flag = foc_khan_strength_intermediate }
                set_global_flag = foc_khan_fleet_scale_large
            }
            else = { set_global_flag = foc_khan_fleet_scale_max }
        }

        # Initial and High had no unique combat multiplier; after the split they
        # both become the Intermediate combat tier, while their fleet quantity
        # is preserved above.
        if = {
            limit = { has_global_flag = foc_khan_strength_initial }
            remove_global_flag = foc_khan_strength_initial
            set_global_flag = foc_khan_strength_intermediate
        }
        if = {
            limit = { has_global_flag = foc_khan_strength_high }
            remove_global_flag = foc_khan_strength_high
            set_global_flag = foc_khan_strength_intermediate
        }

        set_global_flag = foc_khan_three_tier_split_migrated_321
        set_global_flag = foc_khan_strength_migrated_079
        set_global_flag = foc_khan_five_tier_migrated_254
        country_event = { id = foc_khan_settings.1 }
    }
}
