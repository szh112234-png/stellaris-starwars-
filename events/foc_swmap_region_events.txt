# Verify both distinct Jakku gates after the Fallen Empire creation chain.
# The Imperial Remnant gate stays dormant until crisis Ilum supplies its nexus.
country_event = {
    id = foc_swmap.4
    hide_window = yes
    is_triggered_only = yes
    fire_only_once = yes

    trigger = {
        has_global_flag = foc_sw_fallen_empires_ready
        any_system = { has_star_flag = foc_sw_jakku_home }
        any_country = { has_country_flag = foc_sw_imperial_remnant_country }
    }

    immediate = {
        random_country = {
            limit = { has_country_flag = foc_sw_imperial_remnant_country }
            save_global_event_target_as = foc_sw_imperial_remnant
        }
        random_system = {
            limit = { has_star_flag = foc_sw_jakku_home }
            save_global_event_target_as = foc_sw_jakku_system
        }

        event_target:foc_sw_jakku_system = {
            # Jakku-Exegol: the Sith Wayfinder route.
            if = {
                limit = {
                    NOT = {
                        any_system_megastructure = {
                            is_megastructure_type = foc_sith_wayfinder_gate
                        }
                    }
                }
                spawn_megastructure = {
                    type = foc_sith_wayfinder_gate
                    owner = event_target:foc_sw_imperial_remnant
                    planet = star
                    orbit_distance = 230
                    orbit_angle = 300
                    init_effect = {
                        set_name = "NAME_FOC_SITH_WAYFINDER_JAKKU_GATE"
                        set_megastructure_flag = foc_sith_wayfinder_jakku_gate
                        save_global_event_target_as = foc_sith_wayfinder_jakku_gate
                    }
                }
            }
            every_system_megastructure = {
                limit = { is_megastructure_type = foc_sith_wayfinder_gate }
                set_name = "NAME_FOC_SITH_WAYFINDER_JAKKU_GATE"
                set_megastructure_flag = foc_sith_wayfinder_jakku_gate
                save_global_event_target_as = foc_sith_wayfinder_jakku_gate
            }

            # Jakku-Ilum: a separate Imperial Remnant route. Creating its Jakku
            # endpoint now makes both structures visible; the existing Ilum
            # event later creates the counterpart and unlocks actual travel.

                limit = {
                    NOT = {
                        any_system_megastructure = {
                            is_megastructure_type = foc_imperial_remnant_gate
                        }
                    }
                }
                spawn_megastructure = {
                    type = foc_imperial_remnant_gate
                    owner = event_target:foc_sw_imperial_remnant
                    planet = star
                    orbit_distance = 205
                    orbit_angle = 120
                    init_effect = {
                        set_name = "NAME_FOC_IMPERIAL_REMNANT_GATE"
                        set_megastructure_flag = foc_imperial_remnant_jakku_gate
                        save_global_event_target_as = foc_imperial_remnant_jakku_gate
                    }
                }
            }
            every_system_megastructure = {
                limit = { is_megastructure_type = foc_imperial_remnant_gate }
                set_name = "NAME_FOC_IMPERIAL_REMNANT_GATE"
                set_megastructure_flag = foc_imperial_remnant_jakku_gate
                save_global_event_target_as = foc_imperial_remnant_jakku_gate
            }
        }
        set_global_flag = foc_swmap_jakku_wayfinder_verified
        set_global_flag = foc_swmap_jakku_two_gates_verified
        log = "FOC_SWMAP: verified two separate Jakku gates for Exegol and Ilum."
    }
}

# V179: restore Coruscant's original natural main-galaxy wormhole while keeping
# the newly-added Jakku wormhole. The two external systems use different normal
# rim endpoints, so each is an independent route and no hyperlane is created.
country_event = {
    id = foc_swmap.5
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        is_country_type = global_event
        has_global_flag = foc_swmap_simplified_image_layout
        NOT = { has_global_flag = foc_swmap_coruscant_main_wormhole_v179 }
        any_system = { has_star_flag = foc_sw_coruscant_home }
    }

    immediate = {
        if = {
            limit = { NOT = { exists = event_target:foc_sw_coruscant_system } }
            random_system = {
                limit = { has_star_flag = foc_sw_coruscant_home }
                save_global_event_target_as = foc_sw_coruscant_system
            }
        }

        # Reuse a saved valid endpoint if one exists; otherwise choose a fresh
        # ordinary rim system that does not already host Jakku's wormhole.
        if = {
            limit = { NOT = { exists = event_target:foc_swmap_coruscant_wormhole_anchor } }
            random_rim_system = {
                limit = {
                    has_natural_wormhole = no
                    any_neighbor_system = { always = yes }
                    is_fe_cluster = no
                    NOR = {
                        has_star_flag = foc_swmap_main_galaxy_anchor
                        has_star_flag = foc_sw_new_republic_cluster
                        has_star_flag = foc_sw_imperial_remnant_cluster
                        has_star_flag = fop_origin_ilum_home
                        has_star_flag = foc_khan_ilum_home
                        has_star_flag = foc_khan_exegol_home
                        has_star_flag = foc_swmap_v3_external_region
                    }
                }
                set_star_flag = foc_swmap_coruscant_wormhole_anchor
                set_star_flag = foc_sw_coruscant_gateway_neighbor
                save_global_event_target_as = foc_swmap_coruscant_wormhole_anchor
            }
        }

        if = {
            limit = {
                exists = event_target:foc_sw_coruscant_system
                event_target:foc_sw_coruscant_system = { has_natural_wormhole = yes }
            }
            # Old saves may already contain a valid Coruscant wormhole. Do not
            # create a duplicate; simply mark the route as verified.
            set_global_flag = foc_swmap_coruscant_wormhole_fallback
            set_global_flag = foc_swmap_coruscant_main_wormhole_v169
            set_global_flag = foc_swmap_coruscant_main_wormhole_v179
            set_global_flag = foc_sw_coruscant_galaxy_lane_added_311w
            set_global_flag = foc_sw_coruscant_galaxy_lane_verified_311r
            log = "FOC_SWMAP V179: existing Coruscant natural wormhole verified."
        }
        else_if = {
            limit = {
                exists = event_target:foc_sw_coruscant_system
                exists = event_target:foc_swmap_coruscant_wormhole_anchor
                event_target:foc_swmap_coruscant_wormhole_anchor = { has_natural_wormhole = no }
            }
            event_target:foc_swmap_coruscant_wormhole_anchor = {
                spawn_natural_wormhole = { bypass_type = wormhole random_pos = no orbit_angle = 0 }
            }
            event_target:foc_sw_coruscant_system = {
                spawn_natural_wormhole = { bypass_type = wormhole random_pos = no orbit_angle = 180 }
                link_wormholes = event_target:foc_swmap_coruscant_wormhole_anchor
            }
            set_global_flag = foc_swmap_coruscant_wormhole_fallback
            set_global_flag = foc_swmap_coruscant_main_wormhole_v169
            set_global_flag = foc_swmap_coruscant_main_wormhole_v179
            # These only suppress obsolete direct-HYPERLANE repair scripts.

            set_global_flag = foc_sw_coruscant_galaxy_lane_verified_311r
            log = "FOC_SWMAP V179: Coruscant independent main-galaxy wormhole restored alongside Jakku."
        }
        # V182: no recurring Coruscant wormhole repair schedule.
    }
