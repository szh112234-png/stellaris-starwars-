# Spawn the event-only dedicated hull. Its core and every installed Arkship module
# are auto-equipped/upgraded by the normal 4.4 starbase design system.
country_event = {
    id = sai_supremacy_arkship.2
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        is_ai = no
        has_technology = foc_tech_dark_legacy_supremacy
        has_country_flag = sai_supremacy_arkship_pending
        exists = capital_scope
    }
    immediate = {
        capital_scope = { save_event_target_as = sai_supremacy_spawn_planet }
        create_fleet = {
            name = "NAME_SAI_SUPREMACY_ARKSHIP"
            effect = {
                set_owner = root
                if = {
                    limit = { has_global_flag = foc_khan_strength_lowest }
                    create_ship = {
                        fleet_cache_recalc = yes # v17.18h: update Arkship/hero fleet cache at creation.
                        name = "NAME_SAI_SUPREMACY_ARKSHIP"
                        design = "NAME_SAI_SUPREMACY_ARKSHIP_TIER_1_SHELL_FOC_LOWEST"
                        graphical_culture = military_arkship
                        effect = {
                        set_ship_flag = foc_v1718i_arkship_new_spawn

                            if = {
                                limit = {
                                    NOT = { has_global_flag = foc_khan_strength_lowest }
                                }
                                add_modifier = { modifier = sai_supremacy_arkship_stats days = -1 }
                            }
                            set_ship_flag = sai_supremacy_arkship
                            set_ship_flag = sai_supremacy_event_spawned_only
                            ship_event = { id = sai_supremacy_arkship.21 days = 1 }
                        }
                    }
                }
                else = {
                    create_ship = {
                        fleet_cache_recalc = yes # v17.18h: update Arkship/hero fleet cache at creation.
                        name = "NAME_SAI_SUPREMACY_ARKSHIP"
                        design = NAME_SAI_SUPREMACY_ARKSHIP_TIER_1
                        graphical_culture = military_arkship
                        effect = {
                        set_ship_flag = foc_v1718i_arkship_new_spawn

                            if = {
                                limit = {
                                    NOT = { has_global_flag = foc_khan_strength_lowest }
                                }
                                add_modifier = { modifier = sai_supremacy_arkship_stats days = -1 }
                            }
                            set_ship_flag = sai_supremacy_arkship
                            set_ship_flag = sai_supremacy_event_spawned_only
                            ship_event = { id = sai_supremacy_arkship.21 days = 1 }
                        }
                    }
                }
                set_location = { target = event_target:sai_supremacy_spawn_planet distance = 20 angle = random }
            }
        }
        remove_country_flag = sai_supremacy_arkship_pending
        remove_country_flag = sai_supremacy_arkship_granted
    }

# Old-save migration from 1.8-2.2. Change only the starbase size/level.
# Existing Arkship modules and buildings remain installed and immediately rebuild
# against the dedicated Supremacy core instead of being overwritten by a forced whole-design replacement.
country_event = {
	id = sai_supremacy_arkship.15
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		any_owned_ship = {
			has_modifier = sai_supremacy_arkship_stats
			OR = {
				is_ship_size = military_arkship_tier_1
				is_ship_size = military_arkship_tier_2
				is_ship_size = military_arkship_tier_3
			}
		}
	}

	immediate = {
		every_owned_ship = {
			limit = { has_modifier = sai_supremacy_arkship_stats }
			if = {
				limit = { is_ship_size = military_arkship_tier_1 }
				starbase = { set_starbase_size = sai_supremacy_arkship_tier_1 }
			}
			else_if = {
				limit = { is_ship_size = military_arkship_tier_2 }
				starbase = { set_starbase_size = sai_supremacy_arkship_tier_2 }
			}
			else_if = {
				limit = { is_ship_size = military_arkship_tier_3 }
				starbase = { set_starbase_size = sai_supremacy_arkship_tier_3 }
			}
			set_graphical_culture = military_arkship
			set_ship_flag = sai_supremacy_arkship
			set_ship_flag = sai_supremacy_event_spawned_only
			# V186: a nomadic Arkship colony resolves to the carrier ship, not a planet.
			# Identity is preserved by the modifier and ship flags above.
			foc_v177_sync_supremacy_shell_ship = yes
			ship_event = { id = sai_supremacy_arkship.22 days = 1 }
			remove_ship_flag = sai_supremacy_model_tier_1
			remove_ship_flag = sai_supremacy_model_tier_2
			remove_ship_flag = sai_supremacy_model_tier_3
		}
	}
}

# Save cleanup: the permanent modifier remains the authoritative event-ship identity.
country_event = {
	id = sai_supremacy_arkship.13
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		OR = {
			any_owned_ship = {
				OR = {
					OR = { is_ship_size = sai_supremacy_arkship_tier_1 is_ship_size = sai_supremacy_arkship_tier_1_shell_foc_lowest }
					OR = { is_ship_size = sai_supremacy_arkship_tier_2 is_ship_size = sai_supremacy_arkship_tier_2_shell_foc_lowest }
					OR = { OR = { is_ship_size = sai_supremacy_arkship_tier_3 is_ship_size = sai_supremacy_arkship_tier_3_shell_foc_lowest } is_ship_size = sai_supremacy_arkship_tier_3_foc_lowest }
				}
				OR = {
					NOT = { has_modifier = sai_supremacy_arkship_stats }
					NOT = { has_ship_flag = sai_supremacy_arkship }
					NOT = { has_ship_flag = sai_supremacy_event_spawned_only }
					NOT = { has_ship_flag = sai_supremacy_arkship_core_initialized }
					starbase = {
						NOR = {
							has_starbase_module = sai_supremacy_arkship_engineering_bay
							has_starbase_module = sai_supremacy_arkship_beastport
							has_starbase_module = arkship_engineering_bay
							has_starbase_module = arkship_beastport
						}
					}
					starbase = {
						NOR = {
							has_starbase_module = sai_supremacy_arkship_docking_bay
							has_starbase_module = arkship_docking_bay
						}
					}
				}
			}
			any_owned_ship = {
				has_modifier = sai_supremacy_arkship_stats
				NOT = { has_ship_flag = sai_supremacy_event_spawned_only }
			}
			any_owned_ship = {
				has_ship_flag = sai_supremacy_arkship
				NOT = { has_modifier = sai_supremacy_arkship_stats }
			}
			any_owned_ship = { has_modifier = sai_supremacy_hyperspace_tracker }
		}
	}

	immediate = {
		# Upgrade reconstruction can discard flags and installed Arkship modules.
		# The dedicated hull key is authoritative, so restore identity and schedule
		# a one-time core repair on the upgraded ship object.
		every_owned_ship = {
			limit = {
				OR = {
					OR = { is_ship_size = sai_supremacy_arkship_tier_1 is_ship_size = sai_supremacy_arkship_tier_1_shell_foc_lowest }
					OR = { is_ship_size = sai_supremacy_arkship_tier_2 is_ship_size = sai_supremacy_arkship_tier_2_shell_foc_lowest }
					OR = { OR = { is_ship_size = sai_supremacy_arkship_tier_3 is_ship_size = sai_supremacy_arkship_tier_3_shell_foc_lowest } is_ship_size = sai_supremacy_arkship_tier_3_foc_lowest }
				}
			}
			if = {
				limit = { NOT = { has_modifier = sai_supremacy_arkship_stats } }
				if = { limit = { NOT = { has_global_flag = foc_khan_strength_lowest } } add_modifier = { modifier = sai_supremacy_arkship_stats days = -1 } }
			}
			set_ship_flag = sai_supremacy_arkship
			set_ship_flag = sai_supremacy_event_spawned_only
			set_graphical_culture = military_arkship
			# V186: a nomadic Arkship colony resolves to the carrier ship, not a planet.
			# Identity is preserved by the modifier and ship flags above.
			if = {
				limit = {
					OR = {
						NOT = { has_ship_flag = sai_supremacy_arkship_core_initialized }
						starbase = {
							NOR = {
								has_starbase_module = sai_supremacy_arkship_engineering_bay
								has_starbase_module = sai_supremacy_arkship_beastport
								has_starbase_module = arkship_engineering_bay
								has_starbase_module = arkship_beastport
							}
						}
						starbase = {
							NOR = {
								has_starbase_module = sai_supremacy_arkship_docking_bay
								has_starbase_module = arkship_docking_bay
							}
						}
					}
				}
				foc_v177_sync_supremacy_shell_ship = yes
				ship_event = { id = sai_supremacy_arkship.22 days = 1 }
			}
		}

		every_owned_ship = {
			limit = { has_modifier = sai_supremacy_arkship_stats }
			set_ship_flag = sai_supremacy_arkship
			set_ship_flag = sai_supremacy_event_spawned_only
		}
		every_owned_ship = {
			limit = { has_modifier = sai_supremacy_hyperspace_tracker }
			remove_modifier = sai_supremacy_hyperspace_tracker
		}
		every_owned_ship = {
			limit = {
				has_ship_flag = sai_supremacy_arkship
				NOT = { has_modifier = sai_supremacy_arkship_stats }
				NOR = {
					OR = { is_ship_size = sai_supremacy_arkship_tier_1 is_ship_size = sai_supremacy_arkship_tier_1_shell_foc_lowest }
					OR = { is_ship_size = sai_supremacy_arkship_tier_2 is_ship_size = sai_supremacy_arkship_tier_2_shell_foc_lowest }
					OR = { OR = { is_ship_size = sai_supremacy_arkship_tier_3 is_ship_size = sai_supremacy_arkship_tier_3_shell_foc_lowest } is_ship_size = sai_supremacy_arkship_tier_3_foc_lowest }
				}
			}
			remove_ship_flag = sai_supremacy_arkship
			remove_ship_flag = sai_supremacy_event_spawned_only
			remove_ship_flag = sai_supremacy_model_tier_1
			remove_ship_flag = sai_supremacy_model_tier_2
			remove_ship_flag = sai_supremacy_model_tier_3
			remove_ship_flag = sai_supremacy_arkship_core_initialized
		}
	}
}


# Tier III fixed tracker auto-installer.
# set_starbase_module uses the numeric slot index; the seventh slot is the dedicated
# spinal_mount_1 slot declared by the Tier III Supremacy starbase level.
country_event = {
	id = sai_supremacy_arkship.16
	hide_window = yes
	is_triggered_only = yes

	immediate = {
        # Keep the standalone cleanup of obsolete tracker bonuses; do not stack them with the module.
        if = { limit = { has_modifier = sai_supremacy_hyperspace_tracker_country_active } remove_modifier = sai_supremacy_hyperspace_tracker_country_active }
        remove_country_flag = sai_supremacy_hyperspace_tracker_online
        every_owned_ship = {
            limit = { OR = { is_ship_size = sai_supremacy_arkship_tier_1 is_ship_size = sai_supremacy_arkship_tier_1_shell_foc_lowest is_ship_size = sai_supremacy_arkship_tier_2 is_ship_size = sai_supremacy_arkship_tier_2_shell_foc_lowest is_ship_size = sai_supremacy_arkship_tier_3 is_ship_size = sai_supremacy_arkship_tier_3_foc_lowest is_ship_size = sai_supremacy_arkship_tier_3_shell_foc_lowest } }
            if = { limit = { has_modifier = sai_supremacy_hyperspace_tracker } remove_modifier = sai_supremacy_hyperspace_tracker }
            if = { limit = { has_modifier = sai_supremacy_hyperspace_tracker_active } remove_modifier = sai_supremacy_hyperspace_tracker_active }
            if = {
                limit = { OR = { is_ship_size = sai_supremacy_arkship_tier_3 is_ship_size = sai_supremacy_arkship_tier_3_foc_lowest is_ship_size = sai_supremacy_arkship_tier_3_shell_foc_lowest } exists = starbase }
                starbase = {
                    if = {
                        limit = { owner = { has_technology = tech_arkship_tier_3 } NOT = { has_starbase_module = sai_supremacy_hyperspace_tracker_module } }
                        set_starbase_module = { slot = 7 module = sai_supremacy_hyperspace_tracker_module }
                    }
                    if = { limit = { has_starbase_module = sai_supremacy_hyperspace_tracker_module } owner = { set_country_flag = sai_supremacy_hyperspace_tracker_online } }
                }
            }
        }
}
}

# Persistent model replacement for this country's normal Military Arkships.
# It uses an in-place same-tier starbase-size switch; no destruction and no
# initialize_arkship_starbase_effect call, so the existing Arkship colony,
# population, modules and buildings remain attached.
country_event = {
	id = sai_supremacy_arkship.20
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_ai = no
		has_technology = foc_tech_dark_legacy_supremacy
		has_country_flag = sai_supremacy_replace_all_military_arkship_models
		any_owned_ship = {
			OR = {
				is_ship_size = military_arkship_tier_1
				is_ship_size = military_arkship_tier_2
				is_ship_size = military_arkship_tier_3
			}
		}
	}

	immediate = {
		every_owned_ship = {
			limit = {
				OR = {
					is_ship_size = military_arkship_tier_1
					is_ship_size = military_arkship_tier_2
					is_ship_size = military_arkship_tier_3
				}
			}
			if = {
				limit = { is_ship_size = military_arkship_tier_1 }
				starbase = { set_starbase_size = sai_supremacy_arkship_tier_1 }
			}
			else_if = {
				limit = { is_ship_size = military_arkship_tier_2 }
				starbase = { set_starbase_size = sai_supremacy_arkship_tier_2 }
			}
			else_if = {
				limit = { is_ship_size = military_arkship_tier_3 }
				starbase = { set_starbase_size = sai_supremacy_arkship_tier_3 }
			}
			set_graphical_culture = military_arkship
		}
	}
}


# Delayed Arkship initialization. Running this after the fleet has an owner and a
# valid location avoids initializing a half-created starbase object inside create_ship.
ship_event = {
	id = sai_supremacy_arkship.21
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		OR = {
			has_ship_flag = sai_supremacy_event_spawned_only
			has_ship_flag = sai_supremacy_arkship
			has_modifier = sai_supremacy_arkship_stats
		}
	}

	immediate = {
        # Reference: CHANGELOG.zip/events/sai_supremacy_arkship_events.txt, .21.
        # This flag means the native effect was CALLED, not a UI test success.
        if = {
            limit = {
                has_ship_flag = foc_v1718i_arkship_new_spawn
                NOT = { has_ship_flag = foc_v1718i_native_init_called }
                exists = owner
            }
            initialize_arkship_starbase_effect = yes
            set_ship_flag = foc_v1718i_native_init_called
            set_ship_flag = foc_v1718i_fresh_native_modules
            remove_ship_flag = foc_v1718i_arkship_new_spawn
            log = "FOC_V1718I_NATIVE_INIT_CALLED: [This.GetName]; deferred native Arkship setup."
            ship_event = { id = sai_supremacy_arkship.22 days = 1 }
        }
}
}



# v17.18i post-initialization compatibility handler. The native initializer owns
# Arkship setup; this handler never inserts guessed default modules into an old ship.
ship_event = {
	id = sai_supremacy_arkship.22
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		OR = {
			OR = { is_ship_size = sai_supremacy_arkship_tier_1 is_ship_size = sai_supremacy_arkship_tier_1_shell_foc_lowest }
			OR = { is_ship_size = sai_supremacy_arkship_tier_2 is_ship_size = sai_supremacy_arkship_tier_2_shell_foc_lowest }
			OR = { OR = { is_ship_size = sai_supremacy_arkship_tier_3 is_ship_size = sai_supremacy_arkship_tier_3_shell_foc_lowest } is_ship_size = sai_supremacy_arkship_tier_3_foc_lowest }
		}
	}

	immediate = {
        # Source-style invisible module conversion is allowed only immediately
        # after our NEW-ship native initializer. Never repack old player modules.
        if = {
            limit = { has_ship_flag = foc_v1718i_fresh_native_modules exists = starbase }
            owner = { country_event = { id = sai_supremacy_arkship.17 } }
            remove_ship_flag = foc_v1718i_fresh_native_modules
        }
        if = {
            limit = { exists = starbase }
            foc_v1718h_sync_supremacy_build_mode = yes
            if = {
                limit = {
                    has_ship_flag = foc_v1718i_native_init_called
                    starbase = { is_arkship_starbase = yes arkship_starbase_has_shipwright_module = yes }
                }
                set_ship_flag = sai_supremacy_arkship_core_initialized
            }
        }
        else = { log = "FOC_V1718I_STARBASE_MISSING: [This.GetName]; no object recreated." }
        owner = { country_event = { id = foc_v1718i_arkship.10 days = 1 } }
        ship_event = { id = foc_v1718h_arkship.2 days = 1 }
}
}
