namespace = foc_khan_223

# One-time 2.23 repair for old saves created with the 2.22 scope/starbase scripts.
event = {
    id = foc_khan_223.1
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        NOT = { has_global_flag = foc_khan_223_bugfix_migrated }
        OR = {
            any_country = { has_country_flag = foc_khan_first_order_country }
            any_country = { has_country_flag = foc_khan_final_order_country }
            any_country = { has_country_flag = foc_khan_rebel_alliance_country }
            any_country = { has_origin = origin_fop_first_order }
        }
    }

    immediate = {
        set_global_flag = foc_khan_223_bugfix_migrated

        # Repair the playable First Order capital with a real planet scope.
        every_country = {
            limit = { has_origin = origin_fop_first_order }
            random_owned_planet = {
                limit = { is_capital = yes }
                planet = {
                    set_planet_flag = fop_ilum_capital
                    save_global_event_target_as = fop_ilum_planet
                }
            }
        }

        # Repair a missing Rebel capital outpost without requesting a citadel design.
        if = {
            limit = {
                exists = event_target:foc_khan_rebel_alliance
                exists = event_target:foc_khan_rebel_system
            }
            event_target:foc_khan_rebel_system = {
                if = {
                    limit = { NOT = { exists = starbase } }
                    create_starbase = {
                        size = starbase_outpost
                        owner = event_target:foc_khan_rebel_alliance
                        effect = { set_starbase_flag = foc_khan_rebel_base_starbase }
                    }
                }
                else = {
                    starbase = {
                        set_owner = event_target:foc_khan_rebel_alliance
                        set_starbase_flag = foc_khan_rebel_base_starbase
                    }
                }
            }
        }

        # Grant the valid shared crisis authorization and rebuild movement orders.
        every_country = {
            limit = {
                OR = {
                    has_country_flag = foc_khan_first_order_country
                    has_country_flag = foc_khan_final_order_country
                    has_country_flag = foc_khan_rebel_alliance_country
                }
            }
            if = {
                limit = { NOT = { has_technology = foc_tech_swgs_crisis_point_defence } }
                if = { limit = { is_ai = yes } give_technology = { tech = foc_tech_swgs_crisis_point_defence message = no } }
            }
        }

        every_country = {
            limit = {
                has_country_flag = foc_khan_rebel_alliance_country
                NOT = { has_country_flag = foc_story_rebel_surrendered_to_first_order }
            }
            every_owned_fleet = {
                limit = {
                    owner = { is_ai = yes }
                    NOT = { has_fleet_flag = foc_republic_origin_rebellion_garrison }
                    has_fleet_flag = foc_khan_rebel_attack_fleet
                    any_owned_ship = { is_ship_class = shipclass_military }
                }
                remove_auto_move_target = yes
                clear_orders = yes
                clear_fleet_actions = this
                fleet_event = { id = foc_khan_rebel.31 days = 2 random = 6 }
            }
        }

        every_country = {
            limit = { has_country_flag = foc_khan_first_order_country }
            every_owned_fleet = {
                limit = {
                    has_fleet_flag = foc_khan_supremacy_fleet
                    any_owned_ship = { has_ship_flag = foc_khan_supremacy_flagship }
                }
                remove_auto_move_target = yes
                clear_orders = yes
                clear_fleet_actions = this
                fleet_event = { id = foc_khan_supremacy_move.2 days = 3 }
            }
        }

        # Retry systems that were left pending after starbase_ai creation failed.
        every_system = {
            limit = {
                OR = {
                    has_star_flag = foc_khan_pending_first_order_takeover
                    has_star_flag = foc_khan_pending_final_order_takeover
                }
            }
            system_event = { id = foc_khan_expand.2 days = 2 }
        }
    }
}
