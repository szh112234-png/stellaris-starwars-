# --- V115: repair saves where 军阀割据 fired while still the Galactic Empire ----
# The Empire carries foc_sw_imperial_remnant_country for legacy compatibility,
# so older versions could show foc_rem.1 before the Treaty.  Remove only that
# premature state/effect; once the Treaty creates the real Remnant, foc_rem.1
# will be eligible again and will set foc_saga_rem_warlords_valid_v115.
country_event = {
    id = foc_rem.100
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        is_ai = no
        NOT = { has_global_flag = foc_republic_origin_galactic_treaty }
        OR = {
            has_country_flag = foc_republic_origin_player_empire
            has_country_flag = foc_republic_origin_empire_country
        }
        has_country_flag = foc_saga_rem_warlords_shown
        NOT = { has_country_flag = foc_saga_rem_warlords_valid_v115 }
        NOT = { has_country_flag = foc_saga_rem_premature_scrubbed_v115 }
    }

    immediate = {
        remove_country_flag = foc_saga_rem_warlords_shown
        remove_country_flag = foc_saga_rem_chimaera_wait
        remove_modifier = foc_saga_rem_iron_discipline
        remove_modifier = foc_saga_rem_bought_loyalty
        set_country_flag = foc_saga_rem_premature_scrubbed_v115
    }
}

# --- 军阀割据 -----------------------------------------------------------------
country_event = {
    id = foc_rem.1
    title = foc_rem.1.name
    desc = foc_rem.1.desc
    picture = GFX_foc_saga_isd_split
    is_triggered_only = yes

    trigger = {
        is_ai = no
        # V115: the Galactic Empire temporarily carries the legacy Remnant flag
        # for shared ship/component compatibility.  This story must wait until
        # the Treaty has actually converted the country into the Imperial Remnant.
        has_global_flag = foc_republic_origin_galactic_treaty
        has_country_flag = foc_sw_imperial_remnant_country
        NOT = { has_country_flag = foc_republic_origin_empire_country }
        NOT = { has_country_flag = foc_republic_origin_player_empire }
        NOT = { has_country_flag = fop_player_first_order }
        NOT = { has_country_flag = foc_khan_first_order_country }
        # The Remnant line begins only after Thrawn has taken command.
        has_country_flag = foc_saga_thrawn_in_command
        NOT = { has_country_flag = foc_saga_rem_warlords_shown }
    }

    immediate = {
        set_country_flag = foc_saga_rem_warlords_shown
        set_country_flag = foc_saga_rem_warlords_valid_v115
        set_timed_country_flag = { flag = foc_saga_rem_chimaera_wait days = 360 }
    }

    # 铁腕还是收编，代价不同。
    option = {
        name = foc_rem.1.a
        add_modifier = { modifier = foc_saga_rem_iron_discipline days = -1 }
    }
    option = {
        name = foc_rem.1.b
        add_modifier = { modifier = foc_saga_rem_bought_loyalty days = -1 }
    }
}
