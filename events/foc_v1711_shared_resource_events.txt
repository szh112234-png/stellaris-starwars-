# No timer is cleared by a console/legacy early dispatch. Retry while the actual
# timed flag remains; on expiry the valid Rebel country receives the leader.
country_event = {
    id = foc_v1711.20
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        has_global_flag = foc_v1711_leia_join_clock_started
        has_global_flag = foc_republic_origin_empire_war_active
        NOT = { has_global_flag = foc_republic_origin_galactic_treaty }
        NOT = { has_global_flag = foc_v212_leia_joined_imperial_rebels }
    }
    immediate = {
        if = {
            limit = { has_global_flag = foc_v1711_wait_leia_join_10y }
            # Exact scheduled deadline can share a tick with flag expiry.
            country_event = { id = foc_v1711.20 days = 1 }
        }
        else = {
            if = {
                limit = {
                    exists = event_target:foc_khan_rebel_alliance
                    event_target:foc_khan_rebel_alliance = { foc_v1710_is_gcw_rebel = yes }
                }
                event_target:foc_khan_rebel_alliance = { foc_v212_leia_joins_after_alderaan = yes }
            }
            else = {
                random_country = {
                    limit = { foc_v1710_is_gcw_rebel = yes }
                    foc_v212_leia_joins_after_alderaan = yes
                }
            }
        }
    }
}
