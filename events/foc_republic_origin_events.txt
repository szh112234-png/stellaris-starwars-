namespace = foc_republic_origin

# Hidden game-start relocation. The Republic country exists first with the
# engine's temporary ordinary home. Once the 27 unowned Star Wars systems have
# been generated, this event claims all of them in one pass, establishes all
# thirty-one pre-authored member colonies with a sustainable capital building,
# moves the capital to Coruscant, and removes the temporary home. The New
# Republic and Imperial Remnant Fallen Empires are never created on this branch.
country_event = {
    id = foc_republic_origin.1
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        # V197: this relocation remains exclusive to the human Republic origin.
        is_ai = no
        has_origin = origin_foc_republic_unification
        NOT = { has_country_flag = foc_republic_origin_initialized }
        has_global_flag = foc_republic_origin_bootstrap
        exists = event_target:foc_sw_coruscant_system
        exists = event_target:foc_sw_jakku_system
        any_system = { has_star_flag = foc_sw_new_republic_cluster }
        any_system = { has_star_flag = foc_sw_imperial_remnant_cluster }
    }

    immediate = {
        set_update_modifiers_batch = begin
        set_country_flag = foc_republic_origin_initialized
        set_country_flag = foc_republic_origin_republic_country
        set_country_flag = foc_republic_origin_player_republic
        foc_std_weapon_grant_gen_1 = yes
        set_country_flag = foc_republic_origin_order_technology_locked
        # V179: use the exact same vanilla wormhole technology as the playable
        # First Order origin. Grant it directly and reveal the bypass type so
        # both the Coruscant and Jakku natural wormholes are immediately usable.
        give_technology = { tech = tech_wormhole_stabilization message = no }
        add_seen_bypass_type = wormhole
        # Recheck after the full game-start setup has settled.
        save_global_event_target_as = foc_republic_origin_republic
        owner_main_species = { save_global_event_target_as = foc_republic_origin_species }

        capital_scope = {
            planet = {
                save_global_event_target_as = foc_republic_origin_staging_capital
                solar_system = { save_global_event_target_as = foc_republic_origin_staging_system }
            }
        }

        set_name = "FOC_REPUBLIC_ORIGIN_GALACTIC_REPUBLIC"
        change_government = {
            authority = auth_democratic
            cooldown = no
            remove_invalid_civics = yes
            civics = {
                civic = civic_foc_republic_origin_galactic_senate
                civic = civic_foc_republic_origin_defense_fleet
            }
        }
        ruler = {
            set_name = "FOC_REPUBLIC_ORIGIN_CHANCELLOR_NAME"
            change_leader_portrait = foc_chancellor_palpatine_01
            set_leader_tier = leader_tier_legendary
        }
        foc_republic_origin_grant_republic_arsenal = yes

        # Safety cleanup for saves/mod stacks created while this origin still used
        # advanced_start = yes. Keep the player-selected starting fleet and shipset,
        # but the Republic must not begin the campaign with a vanilla Titan.
        every_owned_ship = {
            limit = { is_ship_size = titan }
            destroy_ship = this
        }

        add_modifier = { modifier = foc_republic_origin_union_economy days = -1 }
        add_resource = {
            energy = 1000000
            minerals = 1000000
            food = 500000
            consumer_goods = 500000
            alloys = 1000000
            unity = 1000000
            influence = 1000
            volatile_motes = 50000
            exotic_gases = 50000
            rare_crystals = 50000
            sr_zro = 25000
            sr_dark_matter = 25000
            sr_living_metal = 25000
            nanites = 25000
        }

        # Keep the player-selected ship graphical culture and starting fleet.
        # Republic story fleets spawned later explicitly use republic_01 where needed.

        # Phase one: establish every border before any colony work begins. The
        # region is therefore already Republic territory before the player can
        # enter or inspect one of its systems. No intermediary state exists to
        # be annexed or destroyed.
        every_system = {
            limit = {
                OR = {
                    has_star_flag = foc_sw_new_republic_cluster
                    has_star_flag = foc_sw_imperial_remnant_cluster
                }
            }
            foc_republic_origin_claim_for_republic = yes
        }

        # The Republic-origin bootstrap deliberately skips foc_fallen_war.4,
        # which is the normal event that creates Kuat Drive Yards. Create the
        # completed Mega Shipyard here, after Kuat belongs to the player but
        # before the galaxy is exposed. Bilbringi keeps its separate Imperial
        # shipyard; this is Kuat's own structure and is never relocated.
        if = {
            limit = { NOT = { exists = event_target:foc_sw_kuat_system } }
            random_system = {
                limit = { has_star_flag = foc_sw_kuat_home }
                save_global_event_target_as = foc_sw_kuat_system
            }
        }
        if = {
            limit = { exists = event_target:foc_sw_kuat_system }
            event_target:foc_sw_kuat_system = {
                if = {
                    limit = {
                        NOT = {
                            any_system_megastructure = {
                                OR = {
                                    is_megastructure_type = mega_shipyard_0
                                    is_megastructure_type = mega_shipyard_1
                                    is_megastructure_type = mega_shipyard_2
                                    is_megastructure_type = mega_shipyard_3
                                }
                            }
                        }
                    }
                    spawn_megastructure = {
                        type = mega_shipyard_3
                        owner = root
                        planet = star
                        orbit_distance = 390
                        orbit_angle = 110
                        init_effect = { set_name = "Kuat Drive Yards" }
                    }
                }
                set_star_flag = foc_sw_kuat_mega_shipyard_republic_origin_initialized
            }
        }

        # Outposts hold the borders, but the Republic also needs somewhere to
        # actually build and repair a fleet.
        if = {
            limit = { NOT = { exists = event_target:foc_sw_coruscant_system } }
            random_system = {
                limit = { has_star_flag = foc_sw_coruscant_home }
                save_global_event_target_as = foc_sw_coruscant_system
            }
        }
        foc_republic_origin_establish_core_starbases = yes

        # Phase two: colonize every pre-authored member world. Population stays
        # deliberately small outside Coruscant; the dedicated capital building
        # supplies stability and resources without thirty-one homeworld-sized
        # populations, unemployment pools or duplicated starting industries.
        every_system = {
            limit = {
                OR = {
                    has_star_flag = foc_sw_new_republic_cluster
                    has_star_flag = foc_sw_imperial_remnant_cluster
                }
            }
            every_system_planet = {
                limit = {
                    is_colony = no
                    OR = {
                        has_planet_flag = foc_sw_nr_colony
                        has_planet_flag = foc_sw_ir_colony
                    }
                }
                foc_republic_origin_prepare_member_world = yes
            }
        }

        if = {
            limit = { NOT = { exists = event_target:foc_sw_coruscant_system } }
            random_system = {
                limit = { has_star_flag = foc_sw_coruscant_home }
                save_global_event_target_as = foc_sw_coruscant_system
            }
        }
        event_target:foc_sw_coruscant_system = {
            random_system_planet = {
                limit = { has_planet_flag = foc_sw_coruscant }
                save_global_event_target_as = foc_republic_origin_coruscant
            }
        }

        event_target:foc_republic_origin_staging_capital = {
            every_owned_pop_group = {
                resettle_pop_group = {
                    POP_GROUP = this
                    PLANET = event_target:foc_republic_origin_coruscant
                    PERCENTAGE = 1
                }
            }
        }
        event_target:foc_republic_origin_coruscant = {
            set_owner = root
            set_controller = root
            set_capital = yes
        }

        # Install the sustainable member-world capital only after Coruscant has
        # received the player's original starting population. All thirty-one
        # colonies remain directly controlled and fully available to the player.
        every_system = {
            limit = {
                OR = {
                    has_star_flag = foc_sw_new_republic_cluster
                    has_star_flag = foc_sw_imperial_remnant_cluster
                }
            }
            every_system_planet = {
                limit = {
                    is_colony = yes
                    has_planet_flag = foc_republic_origin_member_world
                }
                foc_republic_origin_install_member_world_nexus = yes
            }
        }
        # V231o: droids are the final capital-population step. No later member-
        # world building initializer is allowed to erase this factory again.
        event_target:foc_republic_origin_coruscant = {
            foc_v231_seed_republic_droid_pops = yes
        }
        every_owned_species = {
            limit = { is_same_value = owner_main_species }
            set_species_homeworld = event_target:foc_republic_origin_coruscant
        }
        every_owned_fleet = {
            limit = { is_mobile = yes }
            set_location = {
                target = event_target:foc_republic_origin_coruscant
                distance = 30
                angle = random
            }
        }

        event_target:foc_republic_origin_staging_system = {
            remove_star_flag = empire_home_system
            remove_star_flag = empire_cluster
            remove_star_flag = foc_republic_origin_staging_system
            if = {
                limit = {
                    exists = starbase
                    starbase.owner = { is_same_value = root }
                }
                starbase = { fleet = { delete_fleet = this } }
            }
        }
        event_target:foc_republic_origin_staging_capital = { remove_planet = yes }

        # Close the shared map bootstrap without advertising the two-Fallen-
        # Empire ready flag. This prevents their war, fleet and crisis keepers
        # from treating the Republic as a post-Imperial setup.
        if = { limit = { exists = event_target:foc_sw_new_republic } clear_global_event_target = foc_sw_new_republic }
        if = { limit = { exists = event_target:foc_sw_imperial_remnant } clear_global_event_target = foc_sw_imperial_remnant }
        remove_global_flag = foc_sw_fallen_empires_initializing
        remove_global_flag = foc_sw_fallen_empires_ready
        set_global_flag = foc_republic_origin_no_jakku_executor_site
        set_global_flag = foc_republic_origin_active
        set_global_flag = foc_khan_crisis_disabled
        remove_global_flag = foc_khan_crisis_spawn_scheduled

        # V72 chronology: the campaign opens on 2200.1.1 with the Naboo Crisis.
        # This is a narrative/political anchor, not a war-forcing date.
        start_situation = {
            type = situation_foc_republic_origin_palpatine_plot
            effect = {
                save_global_event_target_as = foc_republic_origin_palpatine_plot_situation
            }
        }
        start_situation = {
            type = situation_foc_republic_origin_separatist_crisis
            effect = {
                save_global_event_target_as = foc_republic_origin_separatist_crisis_situation
            }
        }

        country_event = { id = foc_republic_origin.2 days = 1 }
        country_event = { id = foc_republic_origin.3 days = 5 }
        country_event = { id = foc_republic_origin.6 days = 12 }
        # V93: Naboo is the master story-clock anchor. The opening settings
        # menu chooses when it fires; every later saga phase is already chained
        # to Naboo or to milestones descended from it.
        set_country_flag = foc_republic_origin_naboo_timeline_pending
        set_update_modifiers_batch = end
    }
}

# Order 66 is the end of the Clone Wars, not an incident during them. Reaching
# the final stage of the conspiracy while the Outer Rim is still in revolt only
# arms the directive; once the Confederacy has surrendered it is dispatched
# without waiting for any later Mandalore story flag.

    id = foc_republic_origin.16
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        has_country_flag = foc_republic_origin_republic_country
        NOT = { has_country_flag = foc_republic_origin_order_66_executed }
        NOT = { has_country_flag = foc_v200_order_66_dispatched }
    }

    immediate = {
        if = {
            limit = {
                has_country_flag = foc_republic_origin_separatist_war_won
                NOT = { has_global_flag = foc_republic_origin_separatist_war_active }
            }
            set_country_flag = foc_v200_order_66_dispatched
            remove_country_flag = foc_republic_origin_order_66_pending
            country_event = { id = foc_republic_origin.25 days = 1 }
        }
        else = {
            # The directive may be armed during the war, but it cannot be issued
            # until the Confederacy has surrendered and the Clone War is over.
            set_country_flag = foc_republic_origin_order_66_pending
        }
    }
}

        save_global_event_target_as = foc_sw_imperial_remnant

        give_technology = { tech = tech_juggernaut message = no }

        # player later orders the Death Star to destroy Jedha.

        event_target:foc_sw_coruscant_system = {
            foc_republic_origin_spawn_death_star_one = yes
        }

            limit = { exists = event_target:foc_khan_rebel_alliance }

                foc_v212_obiwan_trains_luke = yes
            }
        }
        else = {
            random_country = {
                limit = {
                    NOT = { has_country_flag = foc_khan_first_order_resistance_country }
                    OR = {
                        has_country_flag = foc_republic_origin_imperial_rebel_country
                        has_country_flag = foc_republic_origin_rebel_alliance_country
                        has_country_flag = foc_khan_rebel_alliance_country
                    }
                }
                foc_v212_obiwan_trains_luke = yes
            }
        }

        country_event = { id = foc_rogue_one.1 days = 1 }
        if = {
            limit = { is_ai = yes }
            every_country = {
                limit = {
                    is_ai = no
                    has_country_flag = foc_republic_origin_player_rebels
                }
                country_event = { id = foc_republic_origin.169 days = 1 }
            }
        }
    }

    option = { name = foc_republic_origin.170.a }

country_event = {
    id = foc_republic_origin.174
    title = foc_republic_origin.174.name
    desc = foc_republic_origin.174.desc
    picture = GFX_foc_ds2_bothan_death_star_ii
    location = event_target:foc_swmap_endor_system
    is_triggered_only = yes
    force_open = yes

    trigger = {
        has_country_flag = foc_republic_origin_empire_country
        has_global_flag = foc_republic_origin_death_star_one_destroyed
        NOT = { has_country_flag = foc_republic_origin_death_star_two_built }
        NOT = { has_country_flag = foc_republic_origin_death_star_two_destroyed }
        NOT = { has_global_flag = foc_republic_origin_death_star_two_destroyed }
        exists = event_target:foc_swmap_endor_system
        OR = {
            AND = { is_ai = yes has_country_flag = foc_ds2_bothan_revealed }
            AND = { is_ai = no has_country_flag = foc_v138_death_star_two_engineering_work_complete }
        }
    }

    immediate = {
        # Bind creation to this project owner, not an inherited global alias.
        # A queued duplicate callback can acknowledge a real existing hull, but
        # it must never create a second station or fabricate a destruction.
        save_global_event_target_as = foc_republic_origin_empire
        if = {
            limit = { NOT = { any_owned_ship = { has_ship_flag = foc_republic_origin_death_star_two } } }
            event_target:foc_swmap_endor_system = {
                foc_republic_origin_spawn_death_star_two = yes
            }
        }
        # create_ship is synchronous. Commit only after an actual owned DS2
        # exists; otherwise retain completed engineering for the monthly retry.
        if = {
            limit = { any_owned_ship = { has_ship_flag = foc_republic_origin_death_star_two } }

            remove_country_flag = foc_ds2_creation_failed_reported
            if = {
                limit = { has_special_project = FOC_REPUBLIC_ORIGIN_DEATH_STAR_II_PROJECT }
                abort_special_project = { type = FOC_REPUBLIC_ORIGIN_DEATH_STAR_II_PROJECT }
            }
            if = {
                limit = { has_special_project = FOC_V202_DEATH_STAR_II_PROJECT_EMERGENCY }
                abort_special_project = { type = FOC_V202_DEATH_STAR_II_PROJECT_EMERGENCY }
            }
            foc_ds2_reveal_bothan_intelligence = yes
            # V64: Death Star II is one of two Executor unlock routes.
            set_country_flag = foc_republic_origin_executor_research_access
            foc_republic_origin_grant_death_star_two_executor_reward = yes
            foc_ds2_enable_refit_project = yes
            log = "FOC_DS2_COMMISSIONED: verified one owned unfinished Death Star II; construction committed without a Death Star I intermediate hull."
            if = {
                limit = { is_ai = yes }
                every_country = {
                    limit = {
                        is_ai = no
                        has_country_flag = foc_republic_origin_player_rebels
                    }
                    country_event = { id = foc_republic_origin.173 days = 1 }
                }
            }
        }
        else_if = {
            limit = { NOT = { has_country_flag = foc_ds2_creation_failed_reported } }
            set_country_flag = foc_ds2_creation_failed_reported
            log = "FOC_DS2_CREATION_FAILED: no owned Death Star II after creation; built flag and rewards withheld, completed engineering retained for the monthly retry."
        }
    }

    option = { name = foc_republic_origin.174.a }
}
