# 66 号命令之后：穆斯塔法。
country_event = {
    id = foc_padme.12
    title = foc_padme.12.name
    desc = foc_padme.12.desc
    picture = GFX_foc_saga_vader_forged
    # V232: 穆斯塔法是维达戴上面罩的那一刻，改用呼吸声替换原来的通用神秘音效。
    show_sound = foc_v232_vader_breathing
    is_triggered_only = yes

    trigger = {
        foc_v197_republic_story_actor = yes
        has_country_flag = foc_saga_padme_senator_shown
        NOT = { has_country_flag = foc_saga_padme_mustafar_shown }
        OR = {
            has_global_flag = foc_republic_origin_anakin_fallen
            has_global_flag = foc_republic_origin_jedi_purged
        }
    }

    immediate = {
        set_country_flag = foc_saga_padme_mustafar_shown
    }

    option = {
        name = foc_padme.12.a
        set_country_flag = foc_v196_padme_mustafar_confirmed
        set_timed_country_flag = { flag = foc_saga_padme_twins_wait days = 30 }
        add_modifier = { modifier = foc_saga_mustafar_aftermath days = -1 }
        # V231: Mustafar is now purely a story scene. It creates no country,
        # splits no territory, decides no ownership of the Mustafar system and
        # does not gate the birth of the Galactic Empire -- Mustafar passed to
        # the Republic at the Confederate surrender, and the Republic has
        # already become the Empire by the time this scene plays.
        #
        # What it does do is finish Anakin Skywalker. The duel ends, Palpatine
        # recovers what is left of him, and Darth Vader is created here.
        hidden_effect = {
            set_global_flag = foc_v231_mustafar_resolved
            every_country = {
                limit = { has_country_flag = foc_republic_origin_empire_country }
                country_event = { id = foc_republic_origin.240 days = 1 }
            }
        }
    }
}
