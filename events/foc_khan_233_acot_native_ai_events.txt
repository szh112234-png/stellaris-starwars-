namespace = foc_khan_233

# One-time migration for saves which already consumed all older migration flags.
# Transport fleets are returned to native crisis AI. Special Xystons retain
# independent fleets but receive a clean native planet-killer action queue.
event = {
    id = foc_khan_233.1
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        NOT = { has_global_flag = foc_khan_233_acot_native_migrated }
        OR = {
            any_country = { has_country_flag = foc_khan_first_order_country }
            any_country = { has_country_flag = foc_khan_final_order_country }
        }
    }

    immediate = {
        set_global_flag = foc_khan_233_acot_native_migrated

        every_country = {
            limit = {
                OR = {
                    has_country_flag = foc_khan_first_order_country
                    has_country_flag = foc_khan_final_order_country
                }
            }
            every_owned_fleet = {
                limit = { any_owned_ship = { is_ship_class = shipclass_transport } }
                foc_khan_229_transport_fleet_action = yes
                set_fleet_flag = foc_khan_acot_native_transport_233
            }
        }

        every_system = {
            remove_star_flag = foc_khan_xyston_system_reserved_212
            every_system_planet = {
                remove_planet_flag = foc_khan_xyston_target_reserved
            }
        }

        every_country = {
            limit = { has_country_flag = foc_khan_final_order_country }
            every_owned_fleet = {
                limit = { any_owned_ship = { OR = { is_ship_size = foc_khan_xystonsd is_ship_size = foc_khan_xystonsd_foc_lowest } } }
                remove_fleet_flag = foc_khan_xyston_active_single_order_225
                remove_fleet_flag = foc_khan_xyston_cleanup_scheduled_225
                remove_fleet_flag = foc_khan_xyston_retarget_pending_218
                remove_auto_move_target = yes
                clear_orders = yes
                clear_fleet_actions = this
                fleet_event = { id = foc_khan_xyston_ai.10 days = 2 random = 8 }
            }
        }
    }
}
