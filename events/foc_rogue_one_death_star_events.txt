# =============================================================================
# Rogue One / Death Star I story bridge
# =============================================================================
# This chain deliberately reuses the existing Jedha and Scarif state:
#   foc_saga_jedha_destroyed
#   foc_republic_origin_scarif_weakness_obtained
# and therefore does not create a second, competing Death Star weakness system.
# It is player-side aware and old-save safe through one-time country flags.
# =============================================================================

# --- Imperial perspective -----------------------------------------------------

# Death Star completion opens a real decision before any shot is fired.
country_event = {
    id = foc_rogue_one.1
    title = foc_jedha_choice.name
    desc = foc_jedha_choice.desc
    picture = GFX_foc_rogue_one_jedha
    location = event_target:foc_swmap_jedha_system
    is_triggered_only = yes
    force_open = yes

    trigger = {
        has_country_flag = foc_republic_origin_empire_country
        has_country_flag = foc_republic_origin_death_star_one_built
        any_owned_ship = { has_ship_flag = foc_republic_origin_death_star_one }
        NOT = { has_global_flag = foc_republic_origin_death_star_one_destroyed }
        NOT = { has_global_flag = foc_republic_origin_galactic_treaty }
        NOT = { has_country_flag = foc_rogue_one_empire_jedha_shown }
        NOT = { has_global_flag = foc_jedha_choice_presented }
        NOT = { has_global_flag = foc_jedha_choice_resolved }
        NOT = { has_global_flag = foc_saga_jedha_destroyed }
        any_system = {
            has_star_flag = foc_swmap_jedha_home
            any_system_planet = {
                has_planet_flag = foc_swmap_jedha
                NOT = { foc_jedha_is_destroyed_planet = yes }
            }
        }
    }

    immediate = {
        set_global_flag = foc_jedha_choice_presented
        foc_jedha_prepare_targets = yes
    }

    option = {
        name = foc_jedha_choice.destroy
        allow = { any_owned_ship = { has_ship_flag = foc_republic_origin_death_star_one } }
        custom_tooltip = foc_jedha_choice.destroy.tt
        hidden_effect = { foc_jedha_destroy_with_death_star = yes }
        ai_chance = { factor = 100 }
    }
    option = {
        name = foc_jedha_choice.ignore
        custom_tooltip = foc_jedha_choice.ignore.tt
        hidden_effect = { foc_jedha_leave_to_rebels = yes }
        ai_chance = { factor = 0 }
    }
}

# One month after Scarif, if Death Star I was already complete when .14 fired,
# the battlestation drops directly into the Scarif system. The trigger rechecks
# that the same DS-I still exists, so a station destroyed during the 30-day
# delay cannot reappear.
country_event = {
    id = foc_rogue_one.16
    title = foc_rogue_one.16.name
    desc = foc_rogue_one.16.desc
    picture = GFX_foc_rogue_one_scarif_space
    location = event_target:foc_swmap_scarif_system
    is_triggered_only = yes
    force_open = yes
    trigger = {
        foc_v1710_scarif_month_elapsed = yes
        has_country_flag = foc_v1717e_scarif_confirmed
        NOT = {
            has_country_flag = foc_v1717e_scarif_choice_pending
        }
        NOT = {
            has_country_flag = foc_v1717e_emergence_confirmed
        }
        has_global_flag = foc_rogue_one_death_star_emergence_scheduled
        NOT = {
            has_global_flag = foc_rogue_one_death_star_emerged
        }
        foc_v1717e_death_star_one_alive = yes
        exists = event_target:foc_swmap_scarif_system
        OR = {
            is_ai = no
            NOT = {
                any_playable_country = {
                    is_ai = no
                }
            }
        }
    }
    immediate = {
        set_country_flag = foc_v1717e_scarif_sequence_migrated
        set_country_flag = foc_v1717e_scarif_choice_pending
        set_global_flag = foc_rogue_one_death_star_emerged
        event_target:foc_swmap_scarif_system = {
            star = {
                save_global_event_target_as = foc_rogue_one_scarif_star
            }
        }
        if = {
            limit = {
                exists = event_target:foc_republic_origin_death_star_one_system
            }
            event_target:foc_republic_origin_death_star_one_system = {
                remove_star_flag = foc_republic_origin_death_star_one_system
            }
        }
        event_target:foc_republic_origin_empire = {
            every_owned_fleet = {
                limit = {
                    any_owned_ship = {
                        has_ship_flag = foc_republic_origin_death_star_one
                    }
                }
                clear_orders = yes
                set_location = {
                    target = event_target:foc_rogue_one_scarif_star
                    distance = 70
                    angle = 25
                }
                set_fleet_flag = foc_republic_origin_death_star_one_fleet
            }
        }
        event_target:foc_swmap_scarif_system = {
            set_star_flag = foc_republic_origin_death_star_one_system
            save_global_event_target_as = foc_republic_origin_death_star_one_system
        }
    }
    option = {
        name = foc_rogue_one.16.a
        hidden_effect = {
            if = {
                limit = {
                    NOT = {
                        has_country_flag = foc_v1717e_emergence_confirmed
                    }
                }
                if = {
                    limit = {
                        NOT = {
                            has_global_flag = foc_republic_origin_galactic_treaty
                        }
                    }
                    set_country_flag = foc_v1717e_emergence_confirmed
                    country_event = {
                        id = foc_v1710_story.1
                        days = 1
                    }
                }
                remove_country_flag = foc_v1717e_scarif_choice_pending
            }
        }
    }
}
