namespace = foc_v235

# Monthly and load hooks have no country scope. Dispatch once per eligible country.
event = {
    id = foc_v235.0
    hide_window = yes
    is_triggered_only = yes
    immediate = {
        every_country = {
            limit = {
                OR = {
                    foc_v235_is_resistance = yes
                    AND = {
                        is_ai = no
                        OR = {
                            has_country_flag = foc_republic_origin_republic_country
                            has_country_flag = foc_republic_origin_player_empire
                            has_country_flag = foc_republic_origin_player_rebels
                            has_country_flag = foc_republic_origin_player_imperial_remnant
                            has_country_flag = foc_sw_new_republic_country
                        }
                    }
                }
            }
            country_event = { id = foc_v235.1 }
        }
    }
}

country_event = {
    id = foc_v235.1
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        OR = {
            foc_v235_is_resistance = yes
            AND = {
                is_ai = no
                OR = {
                    has_country_flag = foc_republic_origin_republic_country
                    has_country_flag = foc_republic_origin_player_empire
                    has_country_flag = foc_republic_origin_player_rebels
                    has_country_flag = foc_republic_origin_player_imperial_remnant
                    has_country_flag = foc_sw_new_republic_country
                }
            }
        }
    }
    immediate = {
        if = { limit = { foc_v235_is_resistance = yes } foc_v235_ensure_resistance_storage = yes }
        foc_v235_finish_player_treasury_succession = yes
        if = {
            limit = { is_ai = no NOT = { has_country_flag = foc_v235_loaded } }
            set_country_flag = foc_v235_loaded
            log = "FOC_V235_LOADED: treasury continuity, Resistance storage, full opening hull census and Death Star conditional cost loaded."
        }
    }
}

# The exact 900-day callback and the monthly power watcher share built guards.
country_event = {
    id = foc_v235.6
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        is_ai = yes
        has_country_flag = foc_republic_origin_empire_country
        has_country_flag = foc_republic_origin_death_star_project_started
        has_country_flag = foc_v235_ds1_clock_started
        NOT = { has_country_flag = foc_republic_origin_death_star_one_built }
        NOT = { has_global_flag = foc_republic_origin_galactic_treaty }
    }
    immediate = {
        remove_country_flag = foc_v235_ds1_early_wait
        # Check fleet power on the actual deadline even between monthly pulses.
        if = {
            limit = {
                NOT = { has_country_flag = foc_v202_death_star_emergency_acceleration_used }
                any_country = {
                    foc_v235_is_gcw_rebel = yes
                    is_at_war_with = root
                    fleet_power > root.trigger:fleet_power
                }
            }
            set_country_flag = foc_v202_death_star_emergency_acceleration_used
        }
        if = {
            limit = { has_country_flag = foc_v202_death_star_emergency_acceleration_used }
            country_event = { id = foc_republic_origin.170 }
        }
    }
}
