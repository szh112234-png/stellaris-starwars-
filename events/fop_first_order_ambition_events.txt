# Ambition scoring follows Shipgirl's crisis-point pattern. These points are
# displayed by the custom window while narrative objectives gate each rank.
country_event = {
    id = fop_ambition.10
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        has_country_flag = fop_origin_crisis_active
        exists = from
        NOT = { from = { is_same_value = root } }
    }
    immediate = {
        change_variable = { which = fop_force_dominion_points value = 1 }
    }
}

# FOC Phase I: strategic base destruction is an explicit ambition objective.
# This = destroyed starbase; From = destroying fleet.
starbase_event = {
    id = fop_ambition.13
    hide_window = yes
    is_triggered_only = yes
    trigger = { exists = from from = { exists = owner owner = { has_country_flag = fop_origin_crisis_active } } }
    immediate = { from.owner = { change_variable = { which = fop_force_dominion_points value = 10 } } }
}
