namespace = foc_fallen_war

# New-game setup. Like SHIPGIRL, the global event country runs the complete
# hidden initialization chain in the same game-start tick. Both scripted Fallen
# Empires and their clusters therefore already exist when the galaxy becomes playable.
country_event = {
    id = foc_fallen_war.1
    hide_window = yes
    is_triggered_only = yes
    fire_only_once = yes

    trigger = {
        is_country_type = global_event
        NOT = { has_global_flag = foc_sw_fallen_empires_created }
        NOT = { has_global_flag = foc_sw_fallen_empires_initializing }
    }

    immediate = {
        set_global_flag = foc_sw_fallen_empires_initializing
        log = "FOC_SWMAP: generating the image-based Star Wars region; Coruscant is the sole main-galaxy entrance."
        country_event = { id = foc_swmap.1 }
    }
}

country_event = {
    id = foc_fallen_war.41
    hide_window = yes
    is_triggered_only = yes
    fire_only_once = yes

    immediate = {
        set_spawn_system_batch = begin
        no_scope = {
            spawn_system = {
                hyperlane = no
                min_distance >= 235
                max_distance <= 245
                min_orientation_angle = 50
                max_orientation_angle = 55
                initializer = "foc_sw_akiva_system"
            }

            # New Republic capital. 310 degrees follows the same radial direction as
            # the Akiva anchor; the increased distance moves Hosnian farther outward.
            event_target:foc_sw_akiva_system? = {
                spawn_system = {
                    hyperlane = no
                    min_distance >= 32
                    max_distance <= 40
                    min_orientation_angle = 307
                    max_orientation_angle = 313
                    initializer = "foc_sw_hosnian_system"
                }
                # FOC 3.11k: Jakku is farther along the rim and is the sole gateway and
                # bridge shared by the two Fallen Empire clusters.
                spawn_system = {
                    hyperlane = no
                    min_distance >= 60
                    max_distance <= 70
                    min_orientation_angle = 37
                    max_orientation_angle = 43
                    initializer = "foc_sw_jakku_system"
                }
            }

            # Compact New Republic fan: six systems including New Republic Yavin.
            event_target:foc_sw_hosnian_system? = {
                spawn_system = { hyperlane = no min_distance >= 20 max_distance <= 30 min_orientation_angle = 280 max_orientation_angle = 286 initializer = "foc_sw_coruscant_system" }
                spawn_system = { hyperlane = no min_distance >= 20 max_distance <= 30 min_orientation_angle = 307 max_orientation_angle = 313 initializer = "foc_sw_kuat_system" }
                spawn_system = { hyperlane = no min_distance >= 20 max_distance <= 30 min_orientation_angle = 334 max_orientation_angle = 340 initializer = "foc_sw_chandrila_system" }
                spawn_system = { hyperlane = no min_distance >= 20 max_distance <= 30 min_orientation_angle = 253 max_orientation_angle = 259 initializer = "foc_sw_yavin_system" }
                random_system = { limit = { has_star_flag = foc_sw_akiva_home } if = { limit = { NOT = { has_hyperlane_to = PREV } } add_hyperlane = { from = this to = PREV } } }
                random_system = { limit = { has_star_flag = foc_sw_coruscant_home } if = { limit = { NOT = { has_hyperlane_to = PREV } } add_hyperlane = { from = this to = PREV } } }
                random_system = { limit = { has_star_flag = foc_sw_kuat_home } if = { limit = { NOT = { has_hyperlane_to = PREV } } add_hyperlane = { from = this to = PREV } } }
                random_system = { limit = { has_star_flag = foc_sw_chandrila_home } if = { limit = { NOT = { has_hyperlane_to = PREV } } add_hyperlane = { from = this to = PREV } } }
                random_system = { limit = { has_star_flag = foc_sw_yavin_home } if = { limit = { NOT = { has_hyperlane_to = PREV } } add_hyperlane = { from = this to = PREV } } }
            }

            # Jakku alone receives the normal-galaxy lane. Bastion continues farther
            # along the rim away from the New Republic cluster, so every route into
            # either Fallen Empire and every route between them passes through Jakku.
            event_target:foc_sw_jakku_system? = {
                random_system = {
                    limit = { has_star_flag = foc_sw_akiva_home }
                    if = {
                        limit = { NOT = { has_hyperlane_to = PREV } }
                        add_hyperlane = { from = this to = PREV }
                    }
                }
                random_neighbor_system_euclidean = {
                    limit = {
                        NOR = {
                            has_star_flag = foc_sw_new_republic_cluster
                            has_star_flag = foc_sw_imperial_remnant_cluster
                        }
                        NOT = { has_hyperlane_to = PREV }
                        any_neighbor_system_euclidean = { has_hyperlane_to = PREV }
                    }
                    set_star_flag = foc_sw_jakku_gateway_neighbor
                    add_hyperlane = { from = this to = PREV }
                }
                spawn_system = {
                    hyperlane = no
                    min_distance >= 60
                    max_distance <= 70
                    min_orientation_angle = 37
                    max_orientation_angle = 43
                    initializer = "foc_sw_bastion_system"
                }
            }

            # Compact Imperial fan: exactly five systems including Jakku/Bastion.
            event_target:foc_sw_bastion_system? = {
                spawn_system = { hyperlane = no min_distance >= 20 max_distance <= 30 min_orientation_angle = 280 max_orientation_angle = 286 initializer = "foc_sw_bilbringi_system" }
                spawn_system = { hyperlane = no min_distance >= 20 max_distance <= 30 min_orientation_angle = 307 max_orientation_angle = 313 initializer = "foc_sw_yaga_minor_system" }
                spawn_system = { hyperlane = no min_distance >= 20 max_distance <= 30 min_orientation_angle = 334 max_orientation_angle = 340 initializer = "foc_sw_rendili_system" }
                random_system = { limit = { has_star_flag = foc_sw_jakku_home } if = { limit = { NOT = { has_hyperlane_to = PREV } } add_hyperlane = { from = this to = PREV } } }
                random_system = { limit = { has_star_flag = foc_sw_bilbringi_home } if = { limit = { NOT = { has_hyperlane_to = PREV } } add_hyperlane = { from = this to = PREV } } }
                random_system = { limit = { has_star_flag = foc_sw_yaga_minor_home } if = { limit = { NOT = { has_hyperlane_to = PREV } } add_hyperlane = { from = this to = PREV } } }
                random_system = { limit = { has_star_flag = foc_sw_rendili_home } if = { limit = { NOT = { has_hyperlane_to = PREV } } add_hyperlane = { from = this to = PREV } } }
            }
        }
        set_spawn_system_batch = end
        set_global_flag = foc_sw_layout_311d_compact_rim
        set_global_flag = foc_sw_layout_311a_jakku_border
        set_global_flag = foc_sw_layout_311k_jakku_only_gateway
        country_event = { id = foc_fallen_war.2 }
    }
}

                set_owner = event_target:foc_khan_first_order

# FOC 3.11b: rebuild ten combined fleets with ten escorts in every fleet.
country_event = {
    id = foc_fallen_war.54
    hide_window = yes
    is_triggered_only = yes

    # This event used to destroy and respawn all eight of a Fallen Empire's
    # line fleets every single month. Its trigger tested
    # foc_sw_combined_fleets_311b_granted but the immediate never set it, so the
    # one-time 3.11b grant it was written to be never stopped running: the
    # ancient navies teleported home and returned to full strength on every
    # monthly pulse, and grinding one down was impossible by construction.
    #
    # It now does two separate jobs. It performs the original grant exactly
    # once, which still repairs old saves, and after that it replaces losses
    # only when an empire is down to fewer than three line fleets, from an
    # allowance of six for the whole game. The ancient powers stay dangerous
    # and can still be beaten permanently.
    trigger = {
        # The Galactic Treaty applies its full eight-fleet packages once in the
        # origin event. It must never use either this six-use replacement pool
        # or the unlimited .30/.31 loops above.
        NOT = { has_global_flag = foc_republic_origin_galactic_treaty }
        has_country_flag = foc_sw_scripted_fallen_empire
        has_global_flag = foc_sw_fallen_empires_ready
        OR = {
            has_country_flag = foc_sw_new_republic_country
            has_country_flag = foc_sw_imperial_remnant_country
        }
        OR = {
            NOT = { has_country_flag = foc_sw_combined_fleets_311b_granted }
            AND = {
                check_variable = {
                    which = foc_sw_fleet_replacements
                    value < 6
                }
                any_owned_planet = { is_colony = yes }
                count_owned_fleet = {
                    limit = {
                        OR = {
                            has_fleet_flag = foc_sw_new_republic_viscount_fleet
                            has_fleet_flag = foc_sw_imperial_executor_fleet
                        }
                    }
                    count < 3
                }
            }
        }
    }

    immediate = {
        if = {
            # First pass, or a save from before the packages were granted.
            limit = { NOT = { has_country_flag = foc_sw_combined_fleets_311b_granted } }
            set_country_flag = foc_sw_combined_fleets_311b_granted
            set_variable = { which = foc_sw_fleet_replacements value = 0 }
            if = {
                limit = {
                    has_country_flag = foc_sw_new_republic_country
                    exists = event_target:foc_sw_hosnian_system
                    NOT = {
                        any_owned_fleet = { has_fleet_flag = foc_sw_new_republic_viscount_fleet }
                    }
                }
                every_owned_fleet = {
                    limit = { has_fleet_flag = foc_sw_new_republic_nebula_fleet }
                    destroy_fleet = this
                }
                event_target:foc_sw_hosnian_system = {
                    foc_sw_spawn_new_republic_viscount_fleets = yes
                }
            }
            if = {
                limit = {
                    has_country_flag = foc_sw_imperial_remnant_country
                    exists = event_target:foc_sw_bastion_system
                    NOT = {
                        any_owned_fleet = { has_fleet_flag = foc_sw_imperial_executor_fleet }
                    }
                }
                every_owned_fleet = {
                    limit = { has_fleet_flag = foc_sw_imperial_remnant_isd_fleet }
                    destroy_fleet = this
                }
                event_target:foc_sw_bastion_system = {
                    foc_sw_spawn_imperial_remnant_executor_fleets = yes
                }
            }
        }
        else = {
            # Bounded replacement: one fleet, raised at the capital.
            change_variable = {
                which = foc_sw_fleet_replacements
                value = 1
            }
            if = {
                limit = {
                    has_country_flag = foc_sw_new_republic_country
                    exists = event_target:foc_sw_hosnian_system
                }
                event_target:foc_sw_hosnian_system = {
                    foc_sw_spawn_new_republic_replacement_fleet = yes
                }
            }
            else_if = {
                limit = {
                    has_country_flag = foc_sw_imperial_remnant_country
                    exists = event_target:foc_sw_bastion_system
                }
                event_target:foc_sw_bastion_system = {
                    foc_sw_spawn_imperial_remnant_replacement_fleet = yes
                }
            }
        }
    }
}
