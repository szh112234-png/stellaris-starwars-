namespace = foc_khan_208

# One-shot cleanup for saves that passed through 2.04-2.07. It removes all
# persistent manual orbit/move queues from ordinary fleets, restores the stable
# crisis country type, and installs the FOC-only queue on special Xystons.
event = {
	id = foc_khan_208.1
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		NOT = { has_global_flag = foc_khan_208_stability_done }
		NOT = { has_global_flag = foc_khan_crisis_finished }
		OR = {
			has_global_flag = foc_khan_phase_one
			has_global_flag = foc_khan_phase_two
		}
	}
	immediate = {
		set_global_flag = foc_khan_208_stability_done
		every_country = {
			limit = {
				OR = {
					has_country_flag = foc_khan_first_order_country
					has_country_flag = foc_khan_final_order_country
				}
			}
			if = {
				limit = { NOT = { is_country_type = foc_khan_crisis_country } }
				set_country_type = foc_khan_crisis_country
			}

			# Clear old scripted orbit orders once. Afterwards the berserk military
			# minister exclusively controls ordinary combat fleets.
			every_owned_fleet = {
				limit = {
					any_owned_ship = { is_ship_class = shipclass_military }
					NOR = {
						has_fleet_flag = foc_khan_supremacy_fleet
						has_fleet_flag = foc_khan_first_order_engineering_fleet
						has_fleet_flag = foc_khan_final_order_engineering_fleet
					}
				}
				remove_auto_move_target = yes
				clear_orders = yes
				clear_fleet_actions = this
				if = {
					limit = {
						OR = {
							has_fleet_flag = foc_khan_first_order_guard_fleet
							has_fleet_flag = foc_khan_final_order_guard_fleet
						}
					}
					set_fleet_stance = passive
				}
				else = {
					set_fleet_stance = aggressive
				}
			}

			# Special Xystons use a separate persistent queue and are never touched
			# by the ordinary fleet migration again.
			every_owned_fleet = {
				limit = { any_owned_ship = { OR = { is_ship_size = foc_khan_xystonsd is_ship_size = foc_khan_xystonsd_foc_lowest } } }
				remove_fleet_flag = synth_queen_titan_ship
				remove_fleet_flag = foc_khan_final_order_planet_killer_207
				fleet_event = { id = foc_khan_xyston_ai.10 days = 1 }
			}
		}
	}
}
